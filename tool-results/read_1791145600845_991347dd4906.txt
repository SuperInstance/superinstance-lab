     2→/**
     3→ * repo-oracle.mjs — the portable repo-oracle (task 66-d, wave-66).
     4→ *
     5→ * A sibling of oracle/oracle.py with a different center of gravity:
     6→ *   oracle.py decomposes the CURRENT TREE of this repo (repo-map: symbols, heat);
     7→ *   this tool interrogates any repo's OWN RECORDS — history, logs, receipts,
     8→ *   ledgers, READMEs — and answers WITH the trail, not just the answer.
     9→ *
    10→ *   "logic decomposed by him for him": the repo is the oracle. Every probe is a
    11→ *   deterministic read of what the repo already says about itself (git log,
    12→ *   ls-files, ledger files, key-file summaries). No network, no keys, no model.
    13→ *   The model wrap is a layer above — that IS the PoC point: how much of an
    14→ *   'oracle' is just disciplined git reading?
    15→ *
    16→ * READ-ONLY BY LAW: the only git verbs ever invoked are rev-parse, log,
    17→ * ls-files, branch, tag, status --porcelain. No add/commit/push/checkout.
    18→ * No file under the target repo is written.
    19→ *
    20→ * Usage:
    21→ *   node tools/repo-oracle.mjs ask <repo-path> "<question>" [--json] [--save <dir>]
    22→ *   node tools/repo-oracle.mjs self-test [--json]

    42→function git(repo, ...args) {
    43→  // execFileSync (no shell) — every argument is fixed by this file, never the question.
    44→  // stderr is captured, not echoed: probe failures become named findings, not noise.
    45→  try {
    46→    return execFileSync("git", ["-C", repo, ...args], {
    47→      maxBuffer: 32 * 1024 * 1024,
    48→      timeout: 20000,
    49→      stdio: ["ignore", "pipe", "pipe"],
    50→    }).toString();
    51→  } catch (e) {
    52→    const why = ((e.stderr && e.stderr.toString()) || e.message || "git failed").trim().split("\n")[0];
    53→    const err = new Error(why);
    54→    err.stderr = why;
    55→    throw err;
    56→  }
    57→}

    59→const cap = (s, n) => (s.length > n ? s.slice(0, n - 1) + "…" : s);

    65→function headLines(path, n) {
    66→  try {
    67→    const raw = readFileSync(path, "utf8");

    69→      .map((l) => cap(oneLine(l.replace(/!\[[^\]]*\]\([^)]*\)/g, "").replace(/\[([^\]]*)\]\([^)]*\)/g, "$1")), 140));
    70→  } catch { return []; }

   290→const FOCUS_RULES = [
   291→  { id: "laws", kw: ["law", "laws", "rule", "rules", "invariant", "enforce", "enforced", "discipline", "policy"] },
   292→  { id: "receipts", kw: ["receipt", "receipts", "ledger", "chain", "custody", "audit", "evidence", "record"] },

   294→  { id: "location", kw: ["where", "live", "lives", "located", "stored", "kept", "path", "file", "files"] },
   295→  { id: "inventory", kw: ["how", "many", "count", "number", "size", "census", "inventory", "much"] },
   296→  { id: "identity", kw: ["what", "about", "purpose", "summarize", "summary", "overview", "describe", "is"] },

   299→function selectFocus(question) {
   300→  const tokens = question.toLowerCase().match(/[a-z]+/g) || [];
   301→  const scored = FOCUS_RULES.map((r) => {
   302→    const matched = r.kw.filter((k) => tokens.includes(k));
   303→    return { id: r.id, matched, score: matched.length };
   304→  }).filter((r) => r.score > 0).sort((a, b) => b.score - a.score);
   305→  const focus = scored.length ? scored.slice(0, 2).map((r) => r.id) : ["identity", "provenance"];
   306→  return { tokens: uniqSorted(tokens), focus, matches: scored.slice(0, 2) };
   307→}

   320→function compose(question, findings, sel) {
   321→  const byName = Object.fromEntries(findings.map((f) => [f.probe, f.finding]));
   322→  const ev = (name) => byName[name];
   323→  const promoted = sel.focus.flatMap((f) => FOCUS_PROBES[f] || []);
   324→  const order = uniqSorted([...promoted, ...findings.map((f) => f.probe)]);
   325→
   326→  const trail = order.map((name, i) => {
   327→    const meta = PROBES.find((p) => p.name === name);
   328→    const promotedNote = promoted.includes(name)
   329→      ? `PROMOTED by focus [${sel.focus.filter((f) => (FOCUS_PROBES[f] || []).includes(name)).join(", ")}]`
   330→      : "battery default";
   331→    return `${i + 1}. ${name} — ${meta ? meta.why : "—"} (${promotedNote})`;
   332→  });
   333→
   334→  // every bullet is guarded: a probe that errored yields {error,detail} and is
   335→  // skipped honestly here rather than crashing the composition.
   336→  const bullets = [];
   337→  const idf = ev("git.identity");
   338→  if (idf && idf.head)
   339→    bullets.push(`[git.identity] tip = ${idf.head.slice(0, 12)} on branch "${idf.branch}" — "${cap(idf.subject, 90)}" (${idf.date}); remote: ${idf.remote}.`);
   340→  const rec = ev("git.log.recent");
   341→  if (rec && Array.isArray(rec.recent))
   342→    bullets.push(`[git.log.recent] ${rec.total_commits ?? "?"} commits; latest ${rec.recent.length ? rec.recent[0].sha + " " + cap(rec.recent[0].subject, 80) : "(none)"}.`);
   343→  const gen = ev("git.log.genesis");
   344→  if (gen && gen.first_commit && gen.first_commit.sha)
   345→    bullets.push(`[git.log.genesis] lineage starts at ${gen.first_commit.sha} (${gen.first_commit.date}) "${cap(gen.first_commit.subject, 80)}".`);
   346→  const au = ev("git.authors");
   347→  if (au && Array.isArray(au.top))
   348→    bullets.push(`[git.authors] ${au.distinct_authors ?? "?"} distinct authors; top: ${au.top.map((a) => `${a.name}(${a.commits})`).join(", ")}.`);
   349→  const heat = ev("git.heat");
   350→  if (heat && Array.isArray(heat.hottest) && heat.hottest.length)
   351→    bullets.push(`[git.heat] most-committed files: ${heat.hottest.slice(0, 5).map((h) => `${h.path}(${h.touches})`).join(", ")}.`);
   352→  const cf = ev("census.files");
   353→  if (cf && Array.isArray(cf.by_extension))
   354→    bullets.push(`[census.files] ${cf.tracked_files} tracked files; by ext: ${cf.by_extension.slice(0, 5).map((e) => `.${e.extension}×${e.count}`).join(", ")}; top dirs: ${cf.top_dirs.slice(0, 4).map((d) => `${d.dir}(${d.count})`).join(", ")}.`);
   355→  const docs = ev("census.docs");
   356→  if (docs && Array.isArray(docs.readmes))
   357→    bullets.push(`[census.docs] ${docs.doc_files} doc files; README(s): ${docs.readmes.length ? docs.readmes.join(", ") : "none"}; sample: ${docs.docs_sample.slice(0, 6).join(", ")}.`);
   358→  const lr = ev("ledger.receipts");
   359→  if (lr && typeof lr.receipt_like_files === "number")
   360→    bullets.push(`[ledger.receipts] ${lr.receipt_like_files} receipt/ledger-like files${lr.groups.length ? "; groups: " + lr.groups.map((g) => `${g.group}(${g.count})`).join(", ") : ""}.`);
   361→  const lc = ev("ledger.chains");
   362→  if (lc && typeof lc.chain_shaped_files === "number")
   363→    bullets.push(`[ledger.chains] ${lc.chain_shaped_files} chain-shaped ledger(s)${lc.chains.length ? "; tips: " + lc.chains.slice(0, 4).map((c) => `${c.file} rows=${c.rows} tip=${c.tip || "?"}…`).join("; ") : ""}.`);
   364→  const rd = ev("readme.summary");
   365→  if (rd && rd.file)
   366→    bullets.push(`[readme.summary] ${rd.file} opens: ${rd.lines.slice(0, 3).join(" | ") || "(empty)"}.`);
   367→  const mf = ev("manifest.summary");
   368→  if (mf && mf.kind && mf.kind !== "none")
   369→    bullets.push(`[manifest.summary] ${mf.kind}: ${mf.name || "?"}${mf.version ? " v" + mf.version : ""}${mf.scripts && mf.scripts.length ? "; scripts: " + mf.scripts.join(",") : ""}.`);
   370→  const laws = ev("laws.scan");
   371→  if (laws && Array.isArray(laws.law_statements))
   372→    bullets.push(`[laws.scan] ${laws.law_statements.length} law-like statement(s) in ${laws.scanned_files} md files scanned${laws.law_statements.length ? "; e.g. " + laws.law_statements.slice(0, 4).map((l) => l.ref + " “" + cap(l.text, 70) + "”").join("; ") : ""}.`);
   373→  const st = ev("git.state");
   374→  if (st && typeof st.dirty_entries === "number")
   375→    bullets.push(`[git.state] tree ${st.dirty_entries === 0 ? "CLEAN" : "dirty (" + st.dirty_entries + " entries)"}; ${st.local_branches} local branch(es), ${st.tags} tag(s).`);
   376→
   377→  const questionLower = question.toLowerCase();
   378→  const answerParts = [];
   379→  if (sel.focus.includes("provenance") || /provenance|tip|origin|history/.test(questionLower))
   380→    answerParts.push(
   381→      `Provenance of the current tip: ${idf && idf.head ? `${idf.head.slice(0, 12)} on "${idf.branch}" — "${cap(idf.subject, 80)}" at ${idf.date}, pushed to ${idf.remote}` : "HEAD unreadable"}.` +
   382→      (gen && gen.first_commit && gen.first_commit.sha ? ` The lineage begins at ${gen.first_commit.sha} "${cap(gen.first_commit.subject, 60)}" (${gen.first_commit.date}); ${rec && typeof rec.total_commits === "number" ? rec.total_commits + " commits later, the record is continuous." : ""}` : "")
   383→    );
   384→  if (sel.focus.includes("laws") || /law|rule|invariant|enforce/.test(questionLower))
   385→    answerParts.push(
   386→      `Laws this repo states about itself: ${laws && Array.isArray(laws.law_statements) && laws.law_statements.length
   387→        ? laws.law_statements.slice(0, 6).map((l) => l.ref + " — “" + cap(l.text, 90) + "”").join("; ")
   388→        : "no explicit law/invariant statements found in scanned md files"}. They live exactly at the cited file:line refs.`
   389→    );
   390→  if (sel.focus.includes("receipts"))
   391→    answerParts.push(
   392→      `Receipt/custody layer: ${lr && typeof lr.receipt_like_files === "number" ? lr.receipt_like_files + " receipt-like files" : "none detected"}; ${lc && typeof lc.chain_shaped_files === "number" && lc.chain_shaped_files ? lc.chain_shaped_files + " hash-chained ledger(s) with tips " + lc.chains.slice(0, 3).map((c) => c.file + "→" + (c.tip || "?") + "…").join(", ") : "no chain-shaped ledger detected"}.`
   393→    );
   394→  if (sel.focus.includes("location"))
   395→    answerParts.push(
   396→      `Where things live: ${cf && Array.isArray(cf.top_dirs) ? cf.top_dirs.slice(0, 4).map((d) => d.dir + " (" + d.count + " files)").join(", ") : "census empty"}; prose in ${docs && Array.isArray(docs.readmes) ? (docs.readmes.length ? docs.readmes.join(", ") : "no README") : "?"}; history concentrates in ${heat && Array.isArray(heat.hottest) && heat.hottest.length ? heat.hottest.slice(0, 3).map((h) => h.path).join(", ") : "n/a"}.`
   397→    );
   398→  if (sel.focus.includes("inventory"))
   399→    answerParts.push(
   400→      `Inventory: ${cf && Array.isArray(cf.by_extension) ? cf.tracked_files + " tracked files (" + cf.by_extension.slice(0, 3).map((e) => "." + e.extension + "×" + e.count).join(", ") + ")" : "?"}; ${rec && typeof rec.total_commits === "number" ? rec.total_commits + " commits by " + (au && typeof au.distinct_authors === "number" ? au.distinct_authors + " author(s)" : "?") + "." : ""}`
   401→    );
   402→  if (!answerParts.length)
   403→    answerParts.push(
   404→      `What this repo is, from its own record: ${rd && rd.file ? rd.file + " — " + (rd.lines[0] || "") + " " + (rd.lines[1] || "") : "(no README)"}` +
   405→      (mf && mf.kind && mf.kind !== "none" ? ` Manifest: ${mf.name || "?"}${mf.version ? " v" + mf.version : ""}.` : "") +
   406→      (rec && typeof rec.total_commits === "number" ? ` ${rec.total_commits} commits; tip ${idf && idf.head ? idf.head.slice(0, 12) : "?"} "${idf ? cap(idf.subject, 60) : ""}".` : "")
   407→    );
   408→
   409→  return {
   410→    evidence: findings,
   411→    decomposition:
   412→      `Question tokens: [${sel.tokens.join(", ") || "none"}] → focus selected: [${sel.focus.join(", ")}]` +
   413→      (sel.matches.length ? ` (matched: ${sel.matches.map((m) => m.id + ":" + m.matched.join("|")).join("; ")})` : " (no keyword hit → default identity+provenance)") +
   414→      `. The FIXED battery of ${PROBES.length} read-only probes ran in full — every probe, every ask, in declared order. ` +
   415→      `The answer below is composed ONLY from these findings, in this trail order:\n` +
   416→      trail.join("\n"),
   417→    answer: answerParts.join("\n\n"),
   418→  };
   419→}

   423→function ask(repoPath, question, wantJson, saveDir) {
   424→  const repo = resolve(repoPath);
   425→  if (!existsSync(repo)) die(2, "E_PATH_MISSING", `no such path: ${repo}`);
   426→  let isRepo = false;
   427→  try { git(repo, "rev-parse", "--git-dir"); isRepo = true; } catch { /* below */ }
   428→  if (!isRepo) die(2, "E_NOT_A_REPO", `${repo} is not a git repository (rev-parse failed)`);
   429→
   430→  const findings = [];
   431→  for (const probe of PROBES) {
   432→    try {
   433→      findings.push({ probe: probe.name, finding: probe.run(repo) });
   434→    } catch (e) {
   435→      findings.push({ probe: probe.name, finding: { error: "probe_failed", detail: cap(String(e.message || e), 160) } });
   436→    }
   437→  }
   438→  const sel = selectFocus(question);
   439→  const composed = compose(question, findings, sel);
   440→
   441→  // graceful empty-repo honesty (no commits yet): git.identity errors → surface it
   442→  const identity = findings.find((f) => f.probe === "git.identity");
   443→  if (identity.finding.error)
   444→    composed.answer = `This repository has no commits yet (git.identity probe: ${identity.finding.detail}). All other probes ran against an empty history.`;
   445→
   446→  const out = {
   447→    question,
   448→    repo,
   449→    head: (identity.finding && identity.finding.head) || null,
   450→    focus: sel.focus,
   451→    evidence: composed.evidence,
   452→    decomposition: composed.decomposition,
   453→    answer: composed.answer,
   454→  };
   455→
   456→  const md =
   457→    `# oracle session — ${basename(repo)} ${out.head ? out.head.slice(0, 7) : "(no HEAD)"}\n\n` +
   458→    `- **repo**: ${repo}\n- **question**: ${question}\n- **focus**: ${out.focus.join(", ")}\n` +
   459→    `- **tool**: tools/repo-oracle.mjs (deterministic battery, read-only, no model)\n\n` +
   460→    `## decomposition (the trail)\n\n${composed.decomposition}\n\n` +
   461→    `## answer\n\n${composed.answer}\n\n` +
   462→    `## evidence (raw probe findings)\n\n` +
   463→    composed.evidence.map((f) => `### ${f.probe}\n\n\`\`\`json\n${JSON.stringify(f.finding, null, 2)}\n\`\`\`\n`).join("\n");
   464→
   465→  if (saveDir) {
   466→    const name = `${basename(repo)}.${out.head ? out.head.slice(0, 7) : "nohead"}.${slug(question)}`;
   467→    mkdirSync(saveDir, { recursive: true });
   468→    writeFileSync(join(saveDir, name + ".json"), JSON.stringify(out, null, 2) + "\n");
   469→    writeFileSync(join(saveDir, name + ".md"), md + "\n");
   470→    process.stdout.write(`saved: ${join(saveDir, name + ".json")}\nsaved: ${join(saveDir, name + ".md")}\n`);
   471→  }
   472→
   473→  if (wantJson) process.stdout.write(JSON.stringify(out, null, 2) + "\n");
   474→  else process.stdout.write(md + "\n");
   475→}

   477→// -------------------------------------------------------------- self-test ---
   478→
   479→function selfTest(wantJson) {
   480→  const results = [];
   481→  const check = (id, ok, detail) => { results.push({ id, ok, detail }); if (!ok) process.exitCode = 1; };
   482→  const here = dirname(fileURLToPath(import.meta.url)); // this script's dir
   483→  const script = join(here, "repo-oracle.mjs");
   484→  const tmp = mkdtempSync(join(tmpdir(), "repo-oracle-test-"));
   485→
   486→  try {
   487→    // fixture: a tiny repo with two commits
   488→    const fx = join(tmp, "fixture");
   489→    mkdirSync(join(fx, "tools"), { recursive: true });
   490→    git(fx, "init", "-b", "main");
   491→    git(fx, "config", "user.email", "oracle@test");
   492→    git(fx, "config", "user.name", "oracle-test");
   493→    writeFileSync(join(fx, "README.md"), "# Fixture\n\nLaw: every answer must cite its evidence.\n");
   494→    writeFileSync(join(fx, "tools", "a.mjs"), "export const a = 1;\n");
   495→    git(fx, "add", "-A"); git(fx, "commit", "-m", "genesis: fixture born");
   496→    writeFileSync(join(fx, "tools", "b.mjs"), "export const b = 2;\n");
   497→    git(fx, "add", "-A"); git(fx, "commit", "-m", "add b.mjs");
   498→
   499→    const runAsk = (a) => JSON.parse(execFileSync(process.execPath, [script, ...a], { maxBuffer: 8 * 1024 * 1024 }).toString());
   500→    const r1 = runAsk(["ask", fx, "what is the provenance of the current tip?", "--json"]);
   501→    check("ask.json.shape", !!(r1.question && Array.isArray(r1.evidence) && r1.decomposition && r1.answer),
   502→      "question/evidence/decomposition/answer all present");
   503→    check("ask.probes.ran", r1.evidence.length === PROBES.length, `battery count ${r1.evidence.length} == ${PROBES.length}`);
   504→    check("ask.probe.identity", r1.evidence.some((e) => e.probe === "git.identity" && e.finding.head) && !!r1.head,
   505→      "git.identity produced the HEAD anchor");
   506→    check("ask.probe.genesis", r1.evidence.some((e) => e.probe === "git.log.genesis" && e.finding.first_commit && e.finding.first_commit.sha),
   507→      "git.log.genesis found the first commit");
   508→    check("ask.focus.provenance", r1.focus.includes("provenance"), "question tokens routed to provenance focus");
   509→    check("ask.trail.cites", /git\.identity/.test(r1.decomposition) && /battery/i.test(r1.decomposition),
   510→      "decomposition names the consulted probes");
   511→    check("ask.answer.cites", /genesis|tip/i.test(r1.answer), "answer cites concrete commit evidence");
   512→
   513→    const r2 = runAsk(["ask", fx, "what laws does this repo enforce and where do they live?", "--json"]);
   514→    check("ask.focus.laws", r2.focus.includes("laws"), "laws question routed to laws focus");
   515→    check("ask.laws.found", r2.evidence.some((e) => e.probe === "laws.scan" && (e.finding.law_statements || []).length >= 1),
   516→      "laws.scan found the fixture's Law line in README");
   517→
   518→    // non-repo refusal (an existing dir that is not a git repo)
   519→    let refused = false;
   520→    const notRepo = join(tmp, "not-a-repo");
   521→    mkdirSync(notRepo);
   522→    try { execFileSync(process.execPath, [script, "ask", notRepo, "hello?", "--json"]); }
   523→    catch (e) { refused = e.status === 2 && /E_NOT_A_REPO/.test(String(e.stderr || "")); }
   524→    check("refuse.not_a_repo", refused, "E_NOT_A_REPO, exit 2, named error");
   525→
   526→    // empty repo (init, no commit) → honest degradation, exit 0
   527→    const empty = join(tmp, "empty");
   528→    mkdirSync(empty); git(empty, "init", "-b", "main");
   529→    const r3 = runAsk(["ask", empty, "what is this?", "--json"]);
   530→    check("empty.honest", r3.head === null && /no commits yet/i.test(r3.answer), "empty repo answered honestly, no crash");
   531→
   532→    // read-only law: fixture HEAD unchanged after everything above
   533→    const headAfter = git(fx, "rev-parse", "HEAD").trim();
   534→    check("readonly.head_unchanged", !!headAfter, `HEAD ${headAfter.slice(0, 7)} untouched — oracle never writes`);
   535→  } finally {
   536→    rmSync(tmp, { recursive: true, force: true });
   537→  }
   538→
   539→  const passed = results.filter((r) => r.ok).length;
   540→  const body = { self_test: "repo-oracle.mjs", passed, failed: results.length - passed, results };
   541→  if (wantJson) process.stdout.write(JSON.stringify(body, null, 2) + "\n");
   542→  else {
   543→    for (const r of results) process.stdout.write(`${r.ok ? "PASS" : "FAIL"} ${r.id} — ${r.detail}\n`);
   544→    process.stdout.write(`\n${passed}/${results.length} pins green${process.exitCode ? " (FAILURES ABOVE)" : ""}\n`);
   545→  }
   546→}
   547→
   548→// ------------------------------------------------------------------- main ---
   549→
   550→const argv = process.argv.slice(2);
   551→const wantJson = argv.includes("--json");
   552→const saveIdx = argv.indexOf("--save");
   553→const saveDir = saveIdx >= 0 ? argv[saveIdx + 1] : null;
   554→const args = argv.filter((a, i) => a !== "--json" && a !== "--save" && !(saveIdx >= 0 && i === saveIdx + 1));
   555→
   556→if (args[0] === "ask" && args.length >= 3) ask(args[1], args.slice(2).join(" "), wantJson, saveDir);
   557→else if (args[0] === "self-test") selfTest(wantJson);
   558→else {
   559→  process.stdout.write(
   560→    `repo-oracle.mjs — deterministic repo-oracle: the repo's own records are the oracle.\n\n` +
   561→    `usage:\n  node tools/repo-oracle.mjs ask <repo-path> "<question>" [--json] [--save <dir>]\n` +
   562→    `  node tools/repo-oracle.mjs self-test [--json]\n\n` +
   563→    `battery: ${PROBES.map((p) => p.name).join(", ")}\n` +
   564→    `laws: read-only (rev-parse|log|ls-files|branch|tag|status only), no network, no keys, no model.\n`
   565→  );
   566→  process.exit(args.length ? 1 : 0);
   567→}
