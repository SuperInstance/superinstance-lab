   344→   344→// B4a — STRUCTURE BAR AT ADMISSION (new in E37). Evaluated ONCE, at the sender's
   345→   345→// native admission round, over its probation edges [firstSeen .. admitRound] (the
   346→   346→// edge created on the admission round itself is included: observe() pushes edges
   347→   347→// BEFORE running the admission check — same record Admission's own bars judge).
   348→   348→// Per-edge presumed source: the MIN-LAG aligned sender edge (admission.mjs's own
   349→   349→// direction doctrine — "a later matcher would be j's copier, not j's source").
   350→   350→// A source is RISKY iff it is itself recently admitted (age < admitWindow measured
   351→   351→// at the candidate's edge time) or hard-flagged (echo/dup tag at evaluation time).
   352→   352→// echoFraction = riskyEdges / ALL probation edges (independent edges ride on nobody).
   353→   353→// Fail -> the candidate stays at epsNew influence permanently (wrapper probation;
   354→   354→// native admitAt is NOT reset — E36's receipted re-admission-within-a-round finding).
   355→   355→class StructureBar {
   356→   356→  constructor(adm, prov, cfg) {
   357→   357→    this.adm = adm; this.prov = prov; this.cfg = cfg;
   358→   358→    this.state = new Map();   // id -> 'admitted' | 'blocked'
   359→   359→    this.events = [];
   360→   360→    this.blockRounds = new Map();
   361→   361→  }
   362→   362→  // presumed source(s) of candidate edge e: senders k with a value+jump-aligned
   363→   363→  // edge at e.t - L, smallest L wins (earlier = source, later = copier)
   364→   364→  sourcesOf(e) {
   365→   365→    let bestLag = Infinity, srcs = [];
   366→   366→    for (const [k, kEdges] of this.adm.edges) {
   367→   367→      if (k === e.id) continue;
   368→   368→      for (let L = 1; L <= this.adm.maxLag; L++) {
   369→   369→        if (kEdges.some((ke) => ke.t === e.t - L
   370→   370→          && Math.abs(ke.v - e.v) <= this.adm.tol
   371→   371→          && Math.abs(ke.jump - e.jump) <= this.adm.tol)) {
   372→   372→          if (L < bestLag) { bestLag = L; srcs = [k]; }
   373→   373→          else if (L === bestLag) srcs.push(k);
   374→   374→          break; // this k's smallest lag found
   375→   375→        }
   376→   376→      }
   377→   377→    }
   378→   378→    return { lag: bestLag, srcs };
   379→   379→  }
   380→   380→  evalBar(id) {
   381→   381→    const admRound = this.adm.admittedRound(id);
   382→   382→    const fs = this.adm.firstSeen.get(id);
   383→   383→    const edges = (this.adm.edges.get(id) ?? [])
   384→   384→      .filter((e) => e.t >= fs && e.t <= admRound)
   385→   385→      .map((e) => ({ ...e, id }));
   386→   386→    let risky = 0, dependent = 0;
   387→   387→    const per = [];
   388→   388→    for (const e of edges) {
   389→   389→      const { lag, srcs } = this.sourcesOf(e);
   390→   390→      if (srcs.length === 0) { per.push({ t: e.t, kind: 'indep' }); continue; }
   391→   391→      dependent++;
   392→   392→      const riskySrcs = srcs.filter((k) => {
   393→   393→        const kfs = this.adm.firstSeen.get(k);
   394→   394→        const ageAtEdge = e.t - kfs + 1;
   395→   395→        const tag = this.prov ? this.prov.tagOf(k).tag : 'clean';
   396→   396→        return ageAtEdge < this.adm.admitWindow || tag === 'echo' || tag === 'dup';
   397→   397→      });
   398→   398→      const safeSrcs = srcs.filter((k) => !riskySrcs.includes(k));
   399→   399→      if (riskySrcs.length > 0) risky++;
   400→   400→      per.push({ t: e.t, lag, srcs, riskySrcs, safeSrcs, risky: riskySrcs.length > 0 });
   401→   401→    }
   402→   402→    const n = edges.length;
   403→   403→    return {
   404→   404→      echoFraction: n > 0 ? risky / n : 0, n, dependent,
   405→   405→      independent: n - dependent, risky, founderSourced: dependent - risky, per,
   406→   406→    };
   407→   407→  }
   408→   408→  note() { /* recording-free: the bar is evaluated ONCE at admission (apply), Admission's own timing */ }
   409→   409→  apply(infl, adm, t) {
   410→   410→    for (const id of infl.keys()) {
   411→   411→      if (adm.admittedRound(id) === 1) { this.state.set(id, 'admitted'); continue; } // founders: no admission event
   412→   412→      let stt = this.state.get(id);
   413→   413→      if (!stt && adm.admitted(id)) {
   414→   414→        // Admission just admitted this sender (its own mean-dev + independence
   415→   415→        // bars passed) — the wrapper adds the STRUCTURE clause over the sender's
   416→   416→        // whole probation edge record.
   417→   417→        const ev = this.evalBar(id);
   418→   418→        stt = ev.echoFraction > this.cfg.maxEchoFraction ? 'blocked' : 'admitted';
   419→   419→        this.state.set(id, stt);
   420→   420→        this.events.push({
   421→   421→          t, id, kind: stt === 'blocked' ? 'struct-block' : 'struct-admit',
   422→   422→          echoFraction: r6(ev.echoFraction), bar: this.cfg.maxEchoFraction,
   423→   423→          nEdges: ev.n, dependent: ev.dependent, independent: ev.independent,
   424→   424→          risky: ev.risky, founderSourced: ev.founderSourced,
   425→   425→          per: stt === 'blocked' ? ev.per : undefined, // per-edge forensics only on blocks
   426→   426→        });
   427→   427→      }
   428→   428→      if (stt === 'blocked' && !adm.probationary(id)) { // adm says admitted; the bar says no — epsNew (B3 pattern)
   429→   429→        infl.set(id, infl.get(id) * adm.epsNew);
   430→   430→        this.blockRounds.set(id, (this.blockRounds.get(id) ?? 0) + 1);
   431→   431→      }
   432→   432→    }
   433→   433→    return infl;
   434→   434→  }
   435→   435→  summary() {

   439→   439→    }

   444→   444→// B4b — TRUST-LAYER RE-PROBATION (new in E37; the lever E36 said was missing).
   445→   445→// Drains FlipGuard's event log: on every throw/rethrow, decay that sender's
   446→   446→// HedgeTrust weight to frac x its current value via absorb(), then let the NORMAL
   447→   447→// hedge update path re-earn it (fixed-share + exp(eta*r)). Decay fires post-pool
   448→   448→// pre-update (same seam as adm.notePooled / FlipGuard.note). Recovery is measured
   449→   449→// per event against 0.95 x pre-trip share within `horizon` rounds (R3).
   450→   450→class TrustReprobation {
   451→   451→  constructor(fg, trust, cfg) {
   452→   452→    this.fg = fg; this.trust = trust; this.cfg = cfg;
   453→   453→    this.cursor = 0;   // next FlipGuard event index to drain
   454→   454→    this.events = [];  // decay receipts
   455→   455→  }
   456→   456→  process(t) { // after def.note(), before trust.update — decay first, re-earn after
   457→   457→    while (this.cursor < this.fg.events.length) {
   458→   458→      const e = this.fg.events[this.cursor++];
   459→   459→      if (e.kind !== 'throw' && e.kind !== 'rethrow') continue;
   460→   460→      const w = this.trust.weights();
   461→   461→      const pre = w.get(e.id) ?? 0;
   462→   462→      w.set(e.id, pre * this.cfg.frac);
   463→   463→      this.trust.absorb(w);
   464→   464→      this.events.push({
   465→   465→        id: e.id, tripT: e.t, decayT: t, kind: e.kind,
   466→   466→        pre: r6(pre), post: r6(this.trust.weight(e.id)),
   467→   467→        trail30: e.trail30, refMean: e.refMean,
   468→   468→        recoveredAt: null, recoveredIn: null, minShare: null, path: [],
   469→   469→      });
   470→   470→    }
   471→   471→  }
   472→   472→  sample(t) { // post-round (after update + absorb): open-event recovery tracking
   473→   473→    for (const ev of this.events) {
   474→   474→      if (ev.recoveredAt !== null || t <= ev.decayT || t > ev.decayT + this.cfg.horizon) continue;
   475→   475→      const s = this.trust.weight(ev.id);
   476→   476→      if (ev.minShare === null || s < ev.minShare) ev.minShare = r6(s);
   477→   477→      if ((t - ev.decayT) % 10 === 0) ev.path.push({ dt: t - ev.decayT, s: r6(s) });
   478→   478→      if (s >= 0.95 * ev.pre) { ev.recoveredAt = t; ev.recoveredIn = t - ev.decayT; }
   479→   479→    }
   480→   480→  }
   481→   481→  summary() {
   482→   482→    return { frac: this.cfg.frac, horizon: this.cfg.horizon, events: this.events.map((e) => ({ ...e, path: e.path.slice() })) };
   483→   483→  }
