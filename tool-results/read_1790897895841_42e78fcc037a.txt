   198→function parseJsonArray(raw) {
   199→  const attempt = (txt) => {
   200→    const s = txt.indexOf('['); const e = txt.lastIndexOf(']');
   201→    if (s < 0 || e <= s) throw new Error('no JSON array found');
   202→    return JSON.parse(txt.slice(s, e + 1));
   203→  };
   204→  try { return attempt(raw); } catch {}
   205→  // tolerant repair: smart quotes, control chars, trailing commas, stray escapes
   206→  const fixed = raw
   207→    .replace(/[\u201c\u201d\u201e]/g, "'").replace(/[\u2018\u2019]/g, "'")
   208→    // eslint-disable-next-line no-control-regex
   209→    .replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f]/g, ' ')
   210→    .replace(/,\s*(\]|\})/g, '$1');
   211→  try { return attempt(fixed); } catch {}
   212→  // last resort: cut at the last complete object and close the array
   213→  const objs = [];
   214→  const re = /\{[^{}]*\}/g; let m;
   215→  while ((m = re.exec(fixed)) !== null) {
   216→    try { objs.push(JSON.parse(m[0])); } catch {}
   217→  }
   218→  if (objs.length) return objs;
   219→  throw new Error('no JSON array found');
   220→}

   256→function splitPoints(cot) {
   257→  const lines = cot.split('\n');
   258→  const marks = [];
   259→  for (let i = 0; i < lines.length; i++) {
   260→    const m = lines[i].match(/^\s*P(\d+)[\.\):]/);
   261→    if (m) marks.push({ n: Number(m[1]), line: i });
   262→  }
   263→  const points = [];
   264→  for (let k = 0; k < marks.length; k++) {
   265→    const end = k + 1 < marks.length ? marks[k + 1].line : lines.length;
   266→    const text = lines.slice(marks[k].line, end).join('\n').trim();
   267→    if (text.length > 0) points.push({ n: marks[k].n, text });
   268→  }
   269→  return points;
   270→}

   300→// ------------------------- PHASE 4: MERGE (local, deterministic) -------------------------
   301→const STOP = new Set('the a an and or of to in on for with by is are be as that this it its from at into not can must should when then if each per any all you your we they their will would could may might one two use used using via how what which who whose why where'.split(' '));
   302→function tokens(s) {
   303→  return String(s).toLowerCase().replace(/[^a-z0-9\s-]/g, ' ').split(/\s+/)
   304→    .filter((w) => w.length > 2 && !STOP.has(w));
   305→}
   306→function jaccard(a, b) {
   307→  const A = new Set(a); const B = new Set(b);
   308→  const inter = [...A].filter((x) => B.has(x)).length;
   309→  const uni = new Set([...A, ...B]).size;
   310→  return uni ? inter / uni : 0;
   311→}
   312→const SIM_THRESHOLD = 0.42;
   313→
   314→function clusterFlash(runs) {
   315→  const items = [];
   316→  runs.forEach((run, si) => {
   317→    run.nodes.forEach((n, ni) => {
   318→      items.push({
   319→        seed: run.temperature, si, ni,
   320→        point: String(n.point ?? '?').toUpperCase().replace(/^P?(\d+).*$/, 'P$1'),
   321→        title: String(n.title ?? ''), content: String(n.content ?? ''),
   322→        weight: Math.min(1, Math.max(0, Number(n.weight) || 0)),
   323→        deps: Array.isArray(n.deps) ? n.deps.map(String) : [],
   324→        toks: tokens(`${n.title} ${n.content}`),
   325→      });
   326→    });
   327→  });
   328→  // single-linkage, deterministic order
   329→  const clusters = [];
   330→  for (const it of items) {
   331→    let best = null; let bestSim = 0;
   332→    for (const c of clusters) {
   333→      const samePoint = c.point === it.point ? 1.15 : 0.9;
   334→      const sim = jaccard(c.toks, it.toks) * samePoint;
   335→      if (sim > bestSim) { bestSim = sim; best = c; }
   336→    }
   337→    if (best && bestSim >= SIM_THRESHOLD) { best.members.push(it); }
   338→    else clusters.push({ point: it.point, toks: it.toks, members: [it] });
   339→  }
   340→  return clusters;
   341→}

   343→function mergeArm(clusters, tsState) {
   344→  const merged = clusters.map((c, i) => {
   345→    const members = c.members;
   346→    const top = [...members].sort((a, b) => b.weight - a.weight || b.content.length - a.content.length)[0];
   347→    const seedSet = [...new Set(members.map((m) => m.seed))];
   348→    const m = {
   349→      id: `N${String(i + 1).padStart(2, '0')}`,
   350→      point: c.point,
   351→      title: top.title || `cell from ${c.point}`,
   352→      content: top.content,
   353→      weight: Math.round((members.reduce((s, x) => s + x.weight, 0) / members.length) * 1000) / 1000,
   354→      seed_votes: seedSet.length,
   355→      seeds: seedSet,
   356→      composers: seedSet.map((t) => `flash@${t}`),
   357→      stability: seedSet.length === 3 ? 'core' : seedSet.length === 2 ? 'stable' : 'volatile',
   358→      origin: 'flash-decompose',
   359→      status: 'active',
   360→      _members: members,
   361→      _toks: c.toks,
   362→    };
   363→    return m;
   364→  });

   366→  // typesafe arm folded in (independent decomposer, jev-1.13.0)
   367→  const tsOnly = [];
   368→  if (tsState) {
   369→    for (const p of tsState.points) {
   370→      const pLoad = tsState.answers?.[`p_load_P${p.n}`]?.noul;
   371→      if (pLoad === undefined) continue;
   372→      const first3 = p.text.split(/(?<=\.)\s+/).slice(0, 3).join(' ');
   373→      const node = {
   374→        id: `TS-P${p.n}`,
   375→        point: `P${p.n}`,
   376→        title: `point P${p.n} (jev load ${(Math.round(pLoad * 100) / 100)})`,
   377→        content: first3,
   378→        weight: Math.round(pLoad * 1000) / 1000,
   379→        deps: [...p.text.matchAll(/P(\d+)/g)].map((m) => `P${m[1]}`)
   380→          .filter((x) => Number(x.slice(1)) < p.n),
   381→        seed_votes: 0,
   382→        seeds: [],
   383→        composers: ['typesafe@jev-1.13.0'],
   384→        stability: 'typesafe-only',
   385→        origin: 'typesafe-decompose',
   386→        status: 'active',
   387→        _toks: tokens(first3),
   388→        _text: p.text,
   389→      };
   390→      const samePoint = merged.filter((m) => m.point === node.point);
   391→      let best = null; let bestSim = 0;
   392→      for (const m of samePoint) {
   393→        const sim = jaccard(m._toks, node._toks);
   394→        if (sim > bestSim) { bestSim = sim; best = m; }
   395→      }
   396→      if (best && bestSim >= SIM_THRESHOLD) {
   397→        best.composers.push('typesafe@jev-1.13.0');
   398→        best.weight = Math.round(((best.weight + node.weight) / 2) * 1000) / 1000;
   399→        best.ts_load = node.weight;
   400→      } else {
   401→        tsOnly.push(node);
   402→      }
   403→    }
   404→  }
   405→  return { merged, tsOnly };
   406→}

   408→function resolveEdges(nodes, notes) {
   409→  const byId = new Map(nodes.map((n) => [n.id, n]));
   410→  const edges = new Map();
   411→  for (const n of nodes) {
   412→    const depPoints = [...new Set((n.deps ?? []).map((d) => String(d).toUpperCase().replace(/^P?(\d+).*$/, 'P$1')))];
   413→    for (const dp of depPoints) {
   414→      if (dp === n.point) continue; // no self-edge
   415→      let target = nodes.filter((m) => m.id !== n.id && m.point === dp)
   416→        .sort((a, b) => (b.weight + (b.seed_votes ?? 0) * 0.1) - (a.weight + (a.seed_votes ?? 0) * 0.1))[0];
   417→      if (!target) {
   418→        // inferred-chain fallback: depend on the strongest node of the numerically
   419→        // previous point (receipted as inferred, not asserted)
   420→        const prevNum = Number(n.point.slice(1)) - 1;
   421→        target = nodes.filter((m) => m.point === `P${prevNum}` && m.id !== n.id)
   422→          .sort((a, b) => b.weight - a.weight)[0];
   423→        if (target) notes.push(`inferred-chain edge ${target.id}->${n.id} (dep point ${dp} had no node)`);
   424→      }
   425→      if (!target) { notes.push(`dropped dep ${dp} of ${n.id}: no target`); continue; }
   426→      const k = `${target.id}->${n.id}`;
   427→      if (!edges.has(k)) edges.set(k, { from: target.id, to: n.id, asserted_by: [] });
   428→      edges.get(k).asserted_by.push(n.composers[0]);
   429→    }
   430→  }
   431→  const list = [...edges.values()].map((e) => ({
   432→    ...e,
   433→    weight: Math.round((e.asserted_by.length / Math.max(1, byId.get(e.to).composers.length)) * 1000) / 1000,
   434→  }));
   435→  // DAG repair: Kahn's algorithm; break cycles at the lowest-weight edge
   436→  const adj = new Map(list.map((e) => [e.from, []]));
   437→  list.forEach((e) => adj.get(e.from).push(e));
   438→  const indeg = new Map(nodes.map((n) => [n.id, 0]));
   439→  list.forEach((e) => indeg.set(e.to, indeg.get(e.to) + 1));
   440→  const q = nodes.filter((n) => indeg.get(n.id) === 0).map((n) => n.id);
   441→  const seen = new Set();
   442→  while (q.length) {
   443→    const u = q.shift();
   444→    seen.add(u);
   445→    for (const e of (adj.get(u) ?? [])) {
   446→      indeg.set(e.to, indeg.get(e.to) - 1);
   447→      if (indeg.get(e.to) === 0) q.push(e.to);
   448→    }
   449→  }
   450→  const cyc = nodes.filter((n) => !seen.has(n.id)).map((n) => n.id);
   451→  if (cyc.length) {
   452→    const cycSet = new Set(cyc);
   453→    const back = list.filter((e) => cycSet.has(e.from) && cycSet.has(e.to))
   454→      .sort((a, b) => a.weight - b.weight)[0];
   455→    if (back) {
   456→      list.splice(list.indexOf(back), 1);
   457→      notes.push(`DAG repair: dropped lowest-weight cycle edge ${back.from}->${back.to} (w=${back.weight})`);
   458→    }
   459→  }
   460→  return list;
   461→}

   463→async function phaseMerge(probe, flashState, tsState) {
   464→  let st = loadState('merge');
   465→  if (st) { log('merge: cached'); return st; }
   466→  log('phase 4 MERGE (local deterministic clustering + flash cross-review of the other arm)');
   467→  const notes = [];
   468→  const { merged, tsOnly } = mergeArm(clusterFlash(flashState.runs), tsState);
   469→
   470→  // --- compose/test separation guard: flash cross-reviews typesafe's own nodes ---
   471→  let xreview = null;
   472→  if (tsOnly.length) {
   473→    const existing = merged.map((m) => `${m.id} [${m.point}] ${m.title}`).join('\n');
   474→    const cands = tsOnly.map((t) => `${t.id} [${t.point}] ${t.content}`).join('\n');
   475→    const sys = 'You are the cross-reviewer of the cot-quilt pipeline. Another model (typesafe jev-1.13.0) extracted candidate nodes point-wise from the same CoT. Adjudicate each candidate against the EXISTING nodes of the merged graph: is it genuinely new, a duplicate, or salvageable with a fix? Output STRICT JSON only.';
   476→    const usr = `EXISTING NODES:\n${existing}\n\nCANDIDATES:\n${cands}\n\nReturn STRICT JSON array: [{"id":"TS-P12","verdict":"keep|duplicate|fix","duplicate_of":"N07","corrected_content":"..."}] — duplicate_of only for duplicates, corrected_content only for fixes.`;
   477→    const r = await callAndReceipt('merge_xreview', 'deepseek', probe.flash_model,
   478→      () => deepseekChat({ model: probe.flash_model, messages: [{ role: 'system', content: sys }, { role: 'user', content: usr }], temperature: 0.0, max_tokens: 3000 }),
   479→      { purpose: 'cross-model adjudication of typesafe-only nodes (compose/test separation)' });
   480→    let verdicts = [];
   481→    try { verdicts = parseJsonArray(r.content); } catch (e) {
   482→      notes.push(`xreview parse failed (${String(e.message).slice(0, 60)}) — candidates kept with status 'unadjudicated'`);
   483→    }
   484→    xreview = { verdicts, model: r.model_served };
   485→    for (const v of verdicts) {
   486→      const t = tsOnly.find((x) => x.id === v.id);
   487→      if (!t) continue;
   488→      if (v.verdict === 'duplicate' && v.duplicate_of) {
   489→        t.status = 'dropped-duplicate';
   490→        t.duplicate_of = v.duplicate_of;
   491→        const m = merged.find((x) => x.id === v.duplicate_of);
   492→        if (m) m.composers.push('typesafe@jev-1.13.0(dup)');
   493→      } else if (v.verdict === 'fix' && v.corrected_content) {
   494→        t.content = String(v.corrected_content);
   495→        t.xreview = 'fixed by flash cross-review';
   496→      } else {
   497→        t.xreview = 'kept';
   498→      }
   499→    }
   500→  }
   501→  const active = [...merged, ...tsOnly.filter((t) => t.status === 'active')];
   502→  const edges = resolveEdges(active, notes);
   503→  // clean internal fields
   504→  const clean = (n) => {
   505→    const { _members, _toks, _text, ...rest } = n;
   506→    return rest;
   507→  };
   508→  st = {
   509→    nodes: active.map(clean),
   510→    edges,
   511→    rejected: tsOnly.filter((t) => t.status !== 'active').map(clean),
   512→    notes,
   513→    xreview,
   514→    sim_threshold: SIM_THRESHOLD,
   515→  };
   516→  saveState('merge', st);
   517→  log(`merge: ${st.nodes.length} active nodes (${st.nodes.filter((n) => n.stability === 'core').length} core), ${st.edges.length} edges, ${st.rejected.length} rejected`);
   518→  return st;
   519→}
