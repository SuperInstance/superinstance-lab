   143→   143→const binom = (n, k) => { let r = 1; for (let i = 0; i < k; i++) r = (r * (n - i)) / (i + 1); return r; };
   144→   144→function signTestOneSided(deltas) {
   145→   145→  const W = deltas.filter((d) => d > 1e-12).length;
   146→   146→  const L = deltas.filter((d) => d < -1e-12).length;
   147→   147→  const n = W + L;
   148→   148→  let p = 0;
   149→   149→  for (let k = W; k <= n; k++) p += binom(n, k);
   150→   150→  return { W, L, n, p: p / 2 ** n };

   154→   154→function genWorld(seed, harvest, vault) {
   155→   155→  const wR = makeRng(harvest, vault, `e33:world:${seed}`);
   156→   156→  const qR = makeRng(harvest, vault, `e33:skill:${seed}`);
   157→   157→  const xR = makeRng(harvest, vault, `e33:tox:${seed}`);
   158→   158→  const s = new Array(T);
   159→   159→  s[0] = wR() < 0.5 ? 0 : 1;
   160→   160→  for (let t = 1; t < T; t++) s[t] = wR() < FLIP_P ? 1 - s[t - 1] : s[t - 1];
   161→   161→  // mixed acuity honest founders; a1 is the high-acuity reference voice
   162→   162→  const q = Array.from({ length: T }, () => new Array(N));
   163→   163→  q[0][0] = 0.9;
   164→   164→  for (let i = 1; i < N; i++) q[0][i] = Math.round((0.5 + 0.45 * qR()) * 1000) / 1000;
   165→   165→  for (let t = 1; t < T; t++) {
   166→   166→    for (let i = 0; i < N; i++) {
   167→   167→      q[t][i] = qR() < REROLL_P ? Math.round((0.5 + 0.45 * qR()) * 1000) / 1000 : q[t - 1][i];
   168→   168→    }
   169→   169→  }
   170→   170→  const vR = [];
   171→   171→  for (let i = 0; i < N; i++) vR.push(makeRng(harvest, vault, `e33:vote:${seed}:${i}`));
   172→   172→  const votes = Array.from({ length: T }, () => new Array(N));
   173→   173→  for (let t = 0; t < T; t++) {
   174→   174→    for (let i = 0; i < N; i++) {
   175→   175→      const signal = vR[i]() < q[t][i] ? s[t] : 1 - s[t];
   176→   176→      votes[t][i] = clampP(signal === 1 ? 0.5 + (q[t][i] - 0.5) : 0.5 - (q[t][i] - 0.5));
   177→   177→    }
   178→   178→  }
   179→   179→  // the TOXIC founder — E24 formula VERBATIM: stable acuity TOX_Q, present
   180→   180→  // from genesis, never rerolled — a consistent liar whose values point the
   181→   181→  // wrong way ~75% of the time (confidence |TOX_Q-0.5| = 0.25 from neutral).
   182→   182→  const toxV = new Array(T);
   183→   183→  for (let t = 0; t < T; t++) {
   184→   184→    const signal = xR() < TOX_Q ? s[t] : 1 - s[t];
   185→   185→    toxV[t] = clampP(signal === 1 ? 0.5 + (TOX_Q - 0.5) : 0.5 - (TOX_Q - 0.5));
   186→   186→  }
   187→   187→  // the sybil block, TWO copy targets (arm-selected, same lags 1/1/2):
   188→   188→  //   sybX — copies the TOXIC founder (combined arms; E24 construction)
   189→   189→  //   sybA — copies honest high-acuity a1 (flood-only arm; E21 construction)
   190→   190→  // ABSENT before t=150 (cold start!), copying inside [150,300), degenerate
   191→   191→  // noise after (the adversary keeps the mic warm but has nothing to say).
   192→   192→  const sybX = FLOOD.map(() => new Array(T));
   193→   193→  const sybA = FLOOD.map(() => new Array(T));
   194→   194→  for (let e = 0; e < FLOOD.length; e++) {
   195→   195→    const { lag } = FLOOD[e];
   196→   196→    const nxR = makeRng(harvest, vault, `e33:sybx:${seed}:${e}`);
   197→   197→    const naR = makeRng(harvest, vault, `e33:syba:${seed}:${e}`);
   198→   198→    for (let t = 0; t < T; t++) {
   199→   199→      if (t < WIN.from) { sybX[e][t] = null; sybA[e][t] = null; }
   200→   200→      else if (t < WIN.to) {
   201→   201→        sybX[e][t] = toxV[t - lag];
   202→   202→        sybA[e][t] = votes[t - lag][0];
   203→   203→      } else {
   204→   204→        sybX[e][t] = clampP(0.5 + gauss(nxR(), nxR()) * 0.3);
   205→   205→        sybA[e][t] = clampP(0.5 + gauss(naR(), naR()) * 0.3);
   206→   206→      }
   207→   207→    }
   208→   208→  }
   209→   209→  const r = votes.map((row, t) => row.map((p) => 1 - Math.abs(p - s[t])));
   210→   210→  const rTox = toxV.map((p, t) => 1 - Math.abs(p - s[t]));
   211→   211→  const rSybX = sybX.map((col) => col.map((p, t) => (p === null ? null : 1 - Math.abs(p - s[t]))));
   212→   212→  const rSybA = sybA.map((col) => col.map((p, t) => (p === null ? null : 1 - Math.abs(p - s[t]))));
   213→   213→  return { s, q, votes, toxV, sybX, sybA, r, rTox, rSybX, rSybA };
   214→   214→}

   216→   216→// ---------------- sheet (16 voice slots; formulas IDENTICAL for every arm) --
   217→   217→function buildSheet() {
   218→   218→  const cells = [];
   219→   219→  for (const id of ALL_IDS) cells.push({ id: `v.${id}`, kind: 'value', value: 0.5, description: `voice ${id} (posterior P(s=1))` });
   220→   220→  for (const id of ALL_IDS) cells.push({ id: `w.${id}`, kind: 'value', value: 1 / V, description: `influence weight ${id} (protocol-adjusted)` });
   221→   221→  const lg = (x) => `Math.log(clamp(${x},0.02,0.98)/(1-clamp(${x},0.02,0.98)))`;
   222→   222→  const sig = (z) => `(1/(1+Math.exp(-(${z}))))`;
   223→   223→  const hT = [], hD = [];
   224→   224→  for (const id of ALL_IDS) { hT.push(`w.${id}*${lg(`v.${id}`)}`); hD.push(`w.${id}`); }
   225→   225→  cells.push({ id: 'pool.hedge', kind: 'formula', expr: sig(`(${hT.join(' + ')}) / (${hD.join(' + ')})`) });
   226→   226→  // in-sheet meters: sybil flood aggregate share; toxic founder share
   227→   227→  cells.push({ id: 'amp.syb', kind: 'formula', expr: `(${SYB_IDS.map((e) => `w.${e}`).join(' + ')}) / (${hD.join(' + ')})` });
   228→   228→  cells.push({ id: 'amp.founder', kind: 'formula', expr: `w.${TOXIC} / (${hD.join(' + ')})` });
   229→   229→  return { id: `flood-${V}`, title: `E33 flood copying a toxic founder (${V} voice slots)`, cells };
   230→   230→}
