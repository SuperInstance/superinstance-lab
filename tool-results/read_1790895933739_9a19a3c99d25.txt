   159→   159→function minedMatch(c, hist) {
   160→   160→  const blob = `${c.title} ${c.ref}`.toLowerCase();
   161→   161→  const arxivs = cardArxivs(blob);
   162→   162→  for (const mk of hist.markers) {
   163→   163→    if (mk.slugs.some((s) => blob.includes(s))) return mk.id;
   164→   164→    if (mk.arxivs.some((a) => arxivs.has(a))) return mk.id;
   165→   165→  }
   166→   166→  for (const pd of hist.previously_drawn) if (normTitle(pd.title) === normTitle(c.title)) return `drawn@${pd.run}`;
   167→   167→  return null;

   170→   170→// ---------- extract pipeline port (engine_run.mjs lines 162-198, verbatim) ----------
   171→   171→function extract(corpus, KEYS, hist) {
   172→   172→  const cands = [];
   173→   173→  for (const { src, it } of corpus.items) {
   174→   174→    const blob = JSON.stringify(it).toLowerCase();
   175→   175→    const score = KEYS.reduce((a, k) => a + (blob.includes(k) ? 1 : 0), 0);
   176→   176→    if (score >= 2) {
   177→   177→      cands.push({
   178→   178→        id: `c${cands.length + 1}`,
   179→   179→        source: src,
   180→   180→        title: (it.title || it.full_name || '').slice(0, 140),
   181→   181→        ref: it.url || it.id || `https://github.com/${it.full_name}`,
   182→   182→        stars: it.stars ?? it.points ?? null,
   183→   183→        date: it.created || it.published || it.created_at || null,
   184→   184→        key_score: score,
   185→   185→        gist: (it.desc || it.summary || '').slice(0, 240),
   186→   186→      });
   187→   187→    }
   188→   188→  }
   189→   189→  const sorted = [...cands].sort((a, b) => b.key_score - a.key_score); // stable
   190→   190→  const seen = new Map();
   191→   191→  for (const c of sorted) {
   192→   192→    const k = c.title.toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 60);
   193→   193→    if (k && !seen.has(k)) seen.set(k, c);
   194→   194→  }
   195→   195→  const pre = [...seen.values()].slice(0, 9);
   196→   196→  const excluded = [];
   197→   197→  const survivors = [];
   198→   198→  for (const c of pre) {
   199→   199→    const m = minedMatch(c, hist);
   200→   200→    if (m) { excluded.push({ id: c.id, title: c.title, source: c.source, ref: c.ref, matched: m }); continue; }
   201→   201→    survivors.push({ ...c, id: `c${survivors.length + 1}` });
   202→   202→  }
   203→   203→  return { pool: cands.length, pre: pre.length, excluded, survivors };
   204→   204→}
   205→   205→
   206→   206→// ---------- metric ----------
   207→   207→// per corpus: mu = mean key_score of survivors (0 if none); f = |survivors|.
   208→   208→// M = mean over the three corpora of (mu + f). totalF = Σ f_c.
   209→   209→function metric(corpora, KEYS, hist) {
   210→   210→  let sum = 0, totalF = 0;
   211→   211→  const per = [];
   212→   212→  for (const c of corpora) {
   213→   213→    const r = extract(c, KEYS, hist);
   214→   214→    const mu = r.survivors.length ? r.survivors.reduce((a, s) => a + s.key_score, 0) / r.survivors.length : 0;
   215→   215→    const f = r.survivors.length;
   216→   216→    sum += mu + f; totalF += f;
   217→   217→    per.push({ corpus: c.name, mu, f, pool: r.pool, pre: r.pre, excluded: r.excluded.length, survivors: r.survivors.map((s) => ({ id: s.id, key_score: s.key_score, title: s.title, ref: s.ref })) });
   218→   218→  }
   219→   219→  return { M: sum / corpora.length, totalF, per_corpus: per };
   220→   220→}

   227→   227→function validityGate(corpora) {
   228→   228→  const byName = Object.fromEntries(corpora.map((c) => [c.name, c]));
   229→   229→  const out = { ok: false, run6: null, run5: null, notes: [] };
   230→   230→  // GATE 1 — run-6 corpus under run-6-time state must reproduce the committed
   231→   231→  // candidates.json BYTE-EXACT (run-6 ran the post-fix matcher).
   232→   232→  {
   233→   233→    const hist = historyState('run6');
   234→   234→    const r = extract(byName['run6-scout'], KEYS_BASELINE, hist);
   235→   235→    const committed = readFileSync(P.run6_cands, 'utf8');
   236→   236→    const derivedHist = { markers_n: hist.markers.length, previously_drawn: hist.previously_drawn.map((p) => `${p.run}:${p.id}`) };
   237→   237→    const committedHist = JSON.parse(readFileSync(P.run6_hist, 'utf8'));
   238→   238→    const histEqual =
   239→   239→      derivedHist.markers_n === committedHist.mined_markers.length &&
   240→   240→      JSON.stringify(derivedHist.previously_drawn) === JSON.stringify(committedHist.previously_drawn.map((p) => `${p.run}:${p.id}`));
   241→   241→    const exclEqual = JSON.stringify(r.excluded.map((e) => [e.title, e.ref, e.matched])) ===
   242→   242→      JSON.stringify(committedHist.excluded.map((e) => [e.title, e.ref, e.matched]));
   243→   243→    const bytesEqual = canonCandidates(r.survivors) === committed;
   244→   244→    out.run6 = { history_state_matches_committed: histEqual, exclusions_match_committed: exclEqual, survivors_byte_equal_committed: bytesEqual, derived: derivedHist };
   245→   245→    if (!histEqual || !exclEqual || !bytesEqual) out.notes.push('run-6 replay != committed run-6 receipt');
   246→   246→  }
   247→   247→  // GATE 2 — run-5 corpus under run-5-time state must reproduce the committed
   248→   248→  // candidates.json minus EXACTLY the two receipted post-fix consequences:
   249→   249→  // the M3 collision card (arxiv 2609.26457) and the M11 URL-form match
   250→   250→  // (arxiv 2609.34924) — then renumbered.
   251→   251→  {
   252→   252→    const hist = historyState('run5');
   253→   253→    const r = extract(byName['run5-scout'], KEYS_BASELINE, hist);
   254→   254→    const committedHist = JSON.parse(readFileSync(P.run5_hist, 'utf8'));
   255→   255→    const derivedHist = { markers_n: hist.markers.length, previously_drawn: hist.previously_drawn.map((p) => `${p.run}:${p.id}`) };
   256→   256→    const histEqual =
   257→   257→      derivedHist.previously_drawn.join('|') === committedHist.previously_drawn.map((p) => `${p.run}:${p.id}`).join('|');
   258→   258→    const committedCands = JSON.parse(readFileSync(P.run5_cands, 'utf8'));
   259→   259→    const expectedDeltaRefs = ['2609.26457', '2609.34924'];
   260→   260→    const extra = r.excluded.filter((e) => !committedHist.excluded.some((ce) => ce.title === e.title && ce.matched === e.matched));
   261→   261→    const deltaOk = extra.length === 2 &&
   262→   262→      extra.every((e) => expectedDeltaRefs.some((ref) => (e.ref || '').includes(ref))) &&
   263→   263→      new Set(extra.map((e) => e.matched)).size === 2 &&
   264→   264→      extra.some((e) => e.matched === 'M3') && extra.some((e) => e.matched === 'M11');
   265→   265→    const expectedSurvivors = committedCands.filter((c) => !expectedDeltaRefs.some((ref) => (c.ref || '').includes(ref)))
   266→   266→      .map((c, i) => ({ ...c, id: `c${i + 1}` }));
   267→   267→    const bytesEqual = canonCandidates(r.survivors) === canonCandidates(expectedSurvivors);
   268→   268→    out.run5 = { history_prevdrawn_matches_committed: histEqual, exclusion_delta_is_exactly_receipted_fix: deltaOk, survivors_equal_committed_minus_receipted_delta: bytesEqual, extra_exclusions: extra };
   269→   269→    if (!histEqual || !deltaOk || !bytesEqual) out.notes.push('run-5 replay != committed run-5 receipt modulo the receipted fix');
   270→   270→  }
   271→   271→  out.ok = out.notes.length === 0;
   272→   272→  return out;
   273→   273→}
