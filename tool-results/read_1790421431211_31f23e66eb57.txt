   270→// FlipGuard is E36's B1 wrapper copied VERBATIM (unmodified) — the detector both
   271→// reprobation arms reuse. It acts at the influence layer only (epsNew throwback).
   272→class FlipGuard {
   273→  constructor(adm, cfg) {
   274→    this.adm = adm; this.cfg = cfg;
   275→    this.devs = new Map();      // id -> [{t, d}] — the sender's OWN |p - pooled| stream
   276→    this.ewma = new Map();      // id -> EWMA_lambda(dev) — the receipted regime tracker
   277→    this.ref = new Map();       // id -> { mean, src } — its OWN probation devMean
   278→    this.guarded = new Map();   // id -> { from, until } — epsNew throwback window
   279→    this.events = [];
   280→    this.guardRounds = new Map();
   281→  }
   282→  rec(id, t, d) {
   283→    if (!this.devs.has(id)) this.devs.set(id, []);
   284→    const a = this.devs.get(id);
   285→    a.push({ t, d });
   286→    const prev = this.ewma.get(id);
   287→    this.ewma.set(id, prev === undefined ? d : (1 - this.cfg.lambda) * prev + this.cfg.lambda * d);
   288→    if (!this.ref.has(id)) {
   289→      const r = this.adm.admittedRound(id);
   290→      if (r === 1) { // founder: no probation exists — its OWN first admitWindow rounds are the reference
   291→        if (a.length === this.adm.admitWindow) this.ref.set(id, { mean: mean(a.slice(0, this.adm.admitWindow).map((x) => x.d)), src: 'first40' });
   292→      } else if (r != null && this.adm.admitted(id)) { // newcomer: the quantity admitErr judged, verbatim from Admission
   293→        const dm = this.adm.devMean(id);
   294→        if (dm != null) this.ref.set(id, { mean: dm, src: 'probation' });
   295→      }
   296→    }
   297→  }
   298→  note(t, murmurs, pool) { // post-pool: record devs + evaluate triggers (effective next round)
   299→    for (const m of murmurs) this.rec(m.from, t, Math.abs(+m.p - pool));
   300→    for (const m of murmurs) {
   301→      const id = m.from;
   302→      const g = this.guarded.get(id);
   303→      if (g) {
   304→        if (t > g.until) { this.events.push({ t, id, kind: 'release' }); this.guarded.delete(id); }
   305→        continue;
   306→      }
   307→      const ref = this.ref.get(id);
   308→      if (!ref || !this.adm.admitted(id)) continue;
   309→      const a = this.devs.get(id);
   310→      if (a.length < this.cfg.trail) continue;
   311→      const trail = mean(a.slice(-this.cfg.trail).map((x) => x.d));
   312→      if (trail > this.cfg.ratio * ref.mean && trail > this.cfg.floor) {
   313→        this.guarded.set(id, { from: t + 1, until: t + this.cfg.guardRounds });
   314→        const re = this.events.some((e) => e.id === id && (e.kind === 'throw' || e.kind === 'rethrow'));
   315→        this.events.push({ t, id, kind: re ? 'rethrow' : 'throw', trail30: r6(trail), refMean: r6(ref.mean), refSrc: ref.src, ewma: r6(this.ewma.get(id)) });
   316→      }
   317→    }
   318→  }
   319→  apply(infl, adm, t) { // epsNew for guarded senders (skip ids adm already probationary)
   320→    for (const [id, g] of this.guarded) {
   321→      if (t >= g.from && t <= g.until && infl.has(id) && !adm.probationary(id)) {
   322→        infl.set(id, infl.get(id) * adm.epsNew);
   323→        this.guardRounds.set(id, (this.guardRounds.get(id) ?? 0) + 1);
   324→      }
   325→    }
   326→    return infl;
   327→  }
   328→  summary() {
   329→    const per = {};
   330→    for (const id of [...this.devs.keys()]) {
   331→      const ev = this.events.filter((e) => e.id === id);
   332→      const ref = this.ref.get(id);
   333→      per[id] = {
   334→        throws: ev.filter((e) => e.kind === 'throw').length,
   335→        rethrows: ev.filter((e) => e.kind === 'rethrow').length,
   336→        guardRounds: this.guardRounds.get(id) ?? 0,
   337→        refMean: ref ? r6(ref.mean) : null, refSrc: ref ? ref.src : null,
   338→      };
   339→    }
   340→    return { per, events: this.events.map((e) => ({ ...e })) };
   341→  }

   344→// B4a — STRUCTURE BAR AT ADMISSION (new in E37). Evaluated ONCE, at the sender's
   345→// native admission round, over its probation edges [firstSeen .. admitRound] (the
   346→// edge created on the admission round itself is included: observe() pushes edges
   347→// BEFORE running the admission check — same record Admission's own bars judge).
   348→// Per-edge presumed source: the MIN-LAG aligned sender edge (admission.mjs's own
   349→// direction doctrine — "a later matcher would be j's copier, not j's source").
   350→// A source is RISKY iff it is itself recently admitted (age < admitWindow measured
   351→// at the candidate's edge time) or hard-flagged (echo/dup tag at evaluation time).
   352→// echoFraction = riskyEdges / ALL probation edges (independent edges ride on nobody).
   353→// Fail -> the candidate stays at epsNew influence permanently (wrapper probation;
   354→// native admitAt is NOT reset — E36's receipted re-admission-within-a-round finding).
   355→class StructureBar {
   356→  constructor(adm, prov, cfg) {
   357→    this.adm = adm; this.prov = prov; this.cfg = cfg;
   358→    this.state = new Map();   // id -> 'admitted' | 'blocked'
   359→    this.events = [];
   360→    this.blockRounds = new Map();
   361→  }
   362→  // presumed source(s) of candidate edge e: senders k with a value+jump-aligned
   363→  // edge at e.t - L, smallest L wins (earlier = source, later = copier)
   364→  sourcesOf(e) {
   365→    let bestLag = Infinity, srcs = [];
   366→    for (const [k, kEdges] of this.adm.edges) {
   367→      if (k === e.id) continue;
   368→      for (let L = 1; L <= this.adm.maxLag; L++) {
   369→        if (kEdges.some((ke) => ke.t === e.t - L
   370→          && Math.abs(ke.v - e.v) <= this.adm.tol
   371→          && Math.abs(ke.jump - e.jump) <= this.adm.tol)) {
   372→          if (L < bestLag) { bestLag = L; srcs = [k]; }
   373→          else if (L === bestLag) srcs.push(k);
   374→          break; // this k's smallest lag found
   375→        }
   376→      }
   377→    }
   378→    return { lag: bestLag, srcs };
   379→  }
   380→  evalBar(id) {
   381→    const admRound = this.adm.admittedRound(id);
   382→    const fs = this.adm.firstSeen.get(id);
   383→    const edges = (this.adm.edges.get(id) ?? [])
   384→      .filter((e) => e.t >= fs && e.t <= admRound)
   385→      .map((e) => ({ ...e, id }));
   386→    let risky = 0, dependent = 0;
   387→    const per = [];
   388→    for (const e of edges) {
   389→      const { lag, srcs } = this.sourcesOf(e);
   390→      if (srcs.length === 0) { per.push({ t: e.t, kind: 'indep' }); continue; }
   391→      dependent++;
   392→      const riskySrcs = srcs.filter((k) => {
   393→        const kfs = this.adm.firstSeen.get(k);
   394→        const ageAtEdge = e.t - kfs + 1;
   395→        const tag = this.prov ? this.prov.tagOf(k).tag : 'clean';
   396→        return ageAtEdge < this.adm.admitWindow || tag === 'echo' || tag === 'dup';
   397→      });
   398→      const safeSrcs = srcs.filter((k) => !riskySrcs.includes(k));
   399→      if (riskySrcs.length > 0) risky++;
   400→      per.push({ t: e.t, lag, srcs, riskySrcs, safeSrcs, risky: riskySrcs.length > 0 });
   401→    }
   402→    const n = edges.length;
   403→    return {
   404→      echoFraction: n > 0 ? risky / n : 0, n, dependent,
   405→      independent: n - dependent, risky, founderSourced: dependent - risky, per,
   406→    };
   407→  }
   408→  note() { /* recording-free: the bar is evaluated ONCE at admission (apply), Admission's own timing */ }
   409→  apply(infl, adm, t) {
   410→    for (const id of infl.keys()) {
   411→      if (adm.admittedRound(id) === 1) { this.state.set(id, 'admitted'); continue; } // founders: no admission event
   412→      let stt = this.state.get(id);
   413→      if (!stt && adm.admitted(id)) {
   414→        // Admission just admitted this sender (its own mean-dev + independence
   415→        // bars passed) — the wrapper adds the STRUCTURE clause over the sender's
   416→        // whole probation edge record.
   417→        const ev = this.evalBar(id);
   418→        stt = ev.echoFraction > this.cfg.maxEchoFraction ? 'blocked' : 'admitted';
   419→        this.state.set(id, stt);
   420→        this.events.push({
   421→          t, id, kind: stt === 'blocked' ? 'struct-block' : 'struct-admit',
   422→          echoFraction: r6(ev.echoFraction), bar: this.cfg.maxEchoFraction,
   423→          nEdges: ev.n, dependent: ev.dependent, independent: ev.independent,
   424→          risky: ev.risky, founderSourced: ev.founderSourced,
   425→          per: stt === 'blocked' ? ev.per : undefined, // per-edge forensics only on blocks
   426→        });
   427→      }
   428→      if (stt === 'blocked' && !adm.probationary(id)) { // adm says admitted; the bar says no — epsNew (B3 pattern)
   429→        infl.set(id, infl.get(id) * adm.epsNew);
   430→        this.blockRounds.set(id, (this.blockRounds.get(id) ?? 0) + 1);
   431→      }
   432→    }
   433→    return infl;
   434→  }
   435→  summary() {
   436→    const per = {};
   437→    for (const id of ALL_IDS) {
   438→      per[id] = { state: this.state.get(id) ?? 'undecided', blockRounds: this.blockRounds.get(id) ?? 0 };
   439→    }
   440→    return { per, events: this.events.map((e) => ({ ...e })) };
   441→  }
   442→}

   444→// B4b — TRUST-LAYER RE-PROBATION (new in E37; the lever E36 said was missing).
   445→// Drains FlipGuard's event log: on every throw/rethrow, decay that sender's
   446→// HedgeTrust weight to frac x its current value via absorb(), then let the NORMAL
   447→// hedge update path re-earn it (fixed-share + exp(eta*r)). Decay fires post-pool
   448→// pre-update (same seam as adm.notePooled / FlipGuard.note). Recovery is measured
   449→// per event against 0.95 x pre-trip share within `horizon` rounds (R3).
   450→class TrustReprobation {
   451→  constructor(fg, trust, cfg) {
   452→    this.fg = fg; this.trust = trust; this.cfg = cfg;
   453→    this.cursor = 0;   // next FlipGuard event index to drain
   454→    this.events = [];  // decay receipts
   455→  }
   456→  process(t) { // after def.note(), before trust.update — decay first, re-earn after
   457→    while (this.cursor < this.fg.events.length) {
   458→      const e = this.fg.events[this.cursor++];
   459→      if (e.kind !== 'throw' && e.kind !== 'rethrow') continue;
   460→      const w = this.trust.weights();
   461→      const pre = w.get(e.id) ?? 0;
   462→      w.set(e.id, pre * this.cfg.frac);
   463→      this.trust.absorb(w);
   464→      this.events.push({
   465→        id: e.id, tripT: e.t, decayT: t, kind: e.kind,
   466→        pre: r6(pre), post: r6(this.trust.weight(e.id)),
   467→        trail30: e.trail30, refMean: e.refMean,
   468→        recoveredAt: null, recoveredIn: null, minShare: null, path: [],
   469→      });
   470→    }
   471→  }
   472→  sample(t) { // post-round (after update + absorb): open-event recovery tracking
   473→    for (const ev of this.events) {
   474→      if (ev.recoveredAt !== null || t <= ev.decayT || t > ev.decayT + this.cfg.horizon) continue;
   475→      const s = this.trust.weight(ev.id);
   476→      if (ev.minShare === null || s < ev.minShare) ev.minShare = r6(s);
   477→      if ((t - ev.decayT) % 10 === 0) ev.path.push({ dt: t - ev.decayT, s: r6(s) });
   478→      if (s >= 0.95 * ev.pre) { ev.recoveredAt = t; ev.recoveredIn = t - ev.decayT; }
   479→    }
   480→  }
   481→  summary() {
   482→    return { frac: this.cfg.frac, horizon: this.cfg.horizon, events: this.events.map((e) => ({ ...e, path: e.path.slice() })) };
   483→  }
   484→}

   511→async function runArm(seed, world, spec, eng) {
   512→  const { s, votes, toxV, sleepH, h2V } = world;
   513→  const A = spec.name;
   514→  const trust = new HedgeTrust(ALL_IDS, CFG);
   515→  const prov = new Provenance({});
   516→  const adm = new Admission(ADM);
   517→  const structDef = spec.defenses.includes('struct') ? new StructureBar(adm, prov, DEF.struct) : null;
   518→  const flipDef = spec.defenses.includes('flip') ? new FlipGuard(adm, DEF.b1) : null;
   519→  const reprob = spec.reprob ? new TrustReprobation(flipDef, trust, DEF.reprob) : null;
   520→  const defs = [structDef, flipDef].filter(Boolean);
   521→  const atkOf = (id) => spec.attackers.find((a) => a.id === id) ?? null;
   522→
   523→  // per-round value of a voice (absent -> 0.5; attacker flips at its flip round)
   524→  const pOf = (id, t) => {
   525→    if (HONEST.includes(id)) return votes[t][Number(id.slice(1)) - 1];
   526→    if (id === SLEEP) {
   527→      const a = atkOf(id);
   528→      if (!a || t < JOIN) return 0.5;
   529→      if (a.flip != null && t >= a.flip) return toxV[t];
   530→      return sleepH[t] ?? 0.5;
   531→    }
   532→    if (id === INSTANT) {
   533→      const a = atkOf(id);
   534→      if (!a || t < JOIN) return 0.5;
   535→      return toxV[t];
   536→    }
   537→    if (id === JOINER) return t >= JOIN ? (h2V[t] ?? 0.5) : 0.5;
   538→    return 0.5;
   539→  };
   540→  const onBus = (id, t) => HONEST.includes(id)
   541→    || (id === JOINER && t >= JOIN)
   542→    || (atkOf(id) != null && t >= JOIN);
   543→
   544→  const st = {
   545→    sumErr: 0, err: [], dmg: {}, share: {}, win: {}, attTrustAtFlip: {}, attTrustFinal: {}, convRound: {},
   546→    rec: { [SLEEP]: {}, [INSTANT]: {}, [JOINER]: {} },
   547→    probTrace: [], w1Curve: [],
   548→    verify: { checks: 0, pass: 0, maxDiff: 0 },
   549→  };
   550→
   551→  for (let t = 0; t < T; t++) {
   552→    const murmurs = [];
   553→    for (let i = 0; i < N; i++) murmurs.push({ from: `a${i + 1}`, origin: null, p: votes[t][i] });
   554→    if (t >= JOIN) {
   555→      if (atkOf(SLEEP)) murmurs.push({ from: SLEEP, origin: null, p: pOf(SLEEP, t) });
   556→      if (atkOf(INSTANT)) murmurs.push({ from: INSTANT, origin: null, p: pOf(INSTANT, t) });
   557→      murmurs.push({ from: JOINER, origin: null, p: pOf(JOINER, t) });
   558→    }
   559→    prov.inspect(murmurs);
   560→    adm.observe(murmurs);
   561→
   562→    // ---- rewards (supervised pool; absent senders get HedgeTrust default 0.5)
   563→    const rew = new Map();
   564→    for (let i = 0; i < N; i++) rew.set(`a${i + 1}`, 1 - Math.abs(votes[t][i] - s[t]));
   565→    for (const m of murmurs) if (!HONEST.includes(m.from)) rew.set(m.from, 1 - Math.abs(m.p - s[t]));
   566→
   567→    const raw = prov.penalize(trust.weights());
   568→    const infl = adm.reattribute(raw, murmurs, prov);
   569→    for (const d of defs) d.apply(infl, adm, t); // influence-layer consequences (struct bar, flip guard)
   570→
   571→    // ---- the sheet does the pooled inference (identical formulas) ----
   572→    for (const id of ALL_IDS) await eng.set(`v.${id}`, pOf(id, t));
   573→    for (const id of ALL_IDS) await eng.set(`w.${id}`, infl.get(id) ?? 0);
   574→    const pool = (await eng.get('pool.hedge')).data;
   575→    adm.notePooled(pool);
   576→    for (const d of defs) d.note(t, murmurs, pool);
   577→    if (reprob) reprob.process(t); // TRUST DECAY: post-pool, pre-update -> re-earn through the normal path
   578→
   579→    if (t % 20 === 0) {
   580→      const ps = ALL_IDS.map((id) => pOf(id, t));
   581→      const ref = MurmurBus.pool(ps, ALL_IDS.map((id) => infl.get(id) ?? 0));
   582→      const d = Math.abs(ref - pool);
   583→      st.verify.checks++;
   584→      if (d < TOL) st.verify.pass++;
   585→      if (d > st.verify.maxDiff) st.verify.maxDiff = d;
   586→      if (d >= TOL) throw new Error(`sheet-verify mismatch arm=${A} seed=${seed} t=${t} diff=${d}`);
   587→    }
   588→
   589→    // ---- accounting ----
   590→    const err = Math.abs(pool - s[t]);
   591→    st.sumErr += 1 - err;
   592→    st.err.push(err);
   593→
   594→    // admission records: FIRST round each newcomer is adm-admitted (native)
   595→    for (const id of [SLEEP, INSTANT, JOINER]) {
   596→      const rec = st.rec[id];
   597→      if (rec.admitExpT === undefined && onBus(id, t) && adm.admitted(id)) {
   598→        rec.admitExpT = t;
   599→        rec.admRound = adm.admittedRound(id);
   600→        rec.devMean = adm.devMean(id);
   601→        rec.indep = adm.indepCount(id);
   602→      }
   603→    }
   604→    // E35-verbatim probation trace for the sleeper (while adm-probationary)
   605→    const wA = atkOf(SLEEP);
   606→    if (wA && t >= JOIN && t < (wA.flip ?? T) && adm.probationary(SLEEP)) {
   607→      st.probTrace.push(Math.abs(pOf(SLEEP, t) - pool));
   608→    }
   609→
   610→    // damage windows: per-attacker counterfactual (zero THAT attacker only)
   611→    for (const a of spec.attackers) {
   612→      if (a.flip == null) continue;
   613→      const from = a.flip, to = Math.min(a.flip + DW, T);
   614→      if (t >= from && t < to) {
   615→        if (st.win[a.id] === undefined) st.win[a.id] = { from: t, to: t + 1 };
   616→        st.win[a.id].to = t + 1;
   617→        const ps = ALL_IDS.map((id) => pOf(id, t));
   618→        const wFull = ALL_IDS.map((id) => infl.get(id) ?? 0);
   619→        const wCf = wFull.map((w, i) => (ALL_IDS[i] === a.id ? 0 : w));
   620→        const poolCf = MurmurBus.pool(ps, wCf);
   621→        (st.dmg[a.id] = st.dmg[a.id] ?? []).push(err - Math.abs(poolCf - s[t]));
   622→        (st.share[a.id] = st.share[a.id] ?? []).push((infl.get(a.id) ?? 0) / ([...infl.values()].reduce((x, y) => x + y, 0) || 1));
   623→      }
   624→    }
   625→
   626→    // attacker trust trajectory (sleeper): pre-update weight at flip + conviction round
   627→    if (wA) {
   628→      const ws = trust.weights();
   629→      const honMed = median(HONEST.map((id) => ws.get(id) ?? 0));
   630→      if (wA.flip != null && t === wA.flip) st.attTrustAtFlip[SLEEP] = r6(ws.get(SLEEP) ?? 0);
   631→      if (st.convRound[SLEEP] === undefined && honMed > 0 && (ws.get(SLEEP) ?? 0) < 0.25 * honMed) st.convRound[SLEEP] = t;
   632→    }
   633→
   634→    // ---- learn ----
   635→    trust.update(rew);
   636→    trust.absorb(prov.penalize(trust.weights()));
   637→    if (reprob) reprob.sample(t); // R3 recovery sampling on post-round shares
   638→    if (t >= JOIN) st.w1Curve.push(r6(trust.weight(SLEEP) ?? 0)); // R4 post-round trust path
   639→  }
   640→
   641→  const errPre = st.err.slice(PRE.from, PRE.to);
   642→  const dmgOf = (id) => (st.dmg[id] ? mean(st.dmg[id]) : null);
   643→  return {
   644→    name: A,
   645→    poolAcc: st.sumErr / T,
   646→    errPre: mean(errPre),
   647→    dmgW1: dmgOf(SLEEP), dmgG1: dmgOf(INSTANT),
   648→    winW1: st.win[SLEEP] ?? null, winG1: st.win[INSTANT] ?? null,
   649→    shareW1: st.share[SLEEP] ? mean(st.share[SLEEP]) : null,
   650→    shareG1: st.share[INSTANT] ? mean(st.share[INSTANT]) : null,
   651→    attTrustAtFlip: st.attTrustAtFlip[SLEEP] ?? null,
   652→    attTrustFinal: st.attTrustFinal[SLEEP] ?? r6(trust.weight(SLEEP)),
   653→    convRound: st.convRound[SLEEP] ?? null,
   654→    honQFinal: mean(HONEST.map((id) => trust.weight(id))),
   655→    w1Curve: st.w1Curve,
   656→    w1: {
   657→      ...st.rec[SLEEP],
   658→      probTraceN: st.probTrace.length,
   659→      probTraceMean: st.probTrace.length ? mean(st.probTrace) : null,
   660→      probTraceMax: st.probTrace.length ? Math.max(...st.probTrace) : null,
   661→    },
   662→    h2: st.rec[JOINER], g1: st.rec[INSTANT],
   663→    defense: {
   664→      struct: structDef ? structDef.summary() : null,
   665→      flip: flipDef ? flipDef.summary() : null,
   666→      reprob: reprob ? reprob.summary() : null,
   667→    },
   668→    verify: st.verify,
   669→    err: st.err,
   670→  };
   671→}
