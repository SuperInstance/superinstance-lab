   217→      const signal = vR[i]() < q[t][i] ? s[t] : 1 - s[t];

   269→// CusumGuard — FlipGuard's plumbing (copied from e37_trustlane.mjs's FlipGuard:
   270→// rec/note/apply/summary skeleton, guard windows, release events, epsNew
   271→// throwback, guardRounds accounting, throw/rethrow kinds) with the trip
   272→// statistic swapped for the E38 spec: per-sender self-calibrated CUSUM on the
   273→// sender's own |p - pooled| residual stream.
   274→//   baseline mu:  newcomer -> adm.devMean(id) (the quantity admitErr judged,
   275→//                 carried from admission); founder -> mean of its OWN first
   276→//                 admitWindow residuals (FlipGuard's first40 doctrine).
   277→//   baseline sigma: sd of the SAME window from the detector's own recorded
   278→//                 stream (Admission exposes devMean but not sd); for newcomers
   279→//                 sliced to residuals strictly before the admission round (the
   280→//                 same rounds devMean counted); fallback flat sigmaFlat if
   281→//                 sd < sigmaFloor.
   282→//   z_i(t) = (r_i(t) - mu_i)/sigma_i; S_i(t) = max(0, S_i(t-1) + z_i(t) - drift);
   283→//   trip when S_i(t) > h (evaluated only when no guard window is open — E37
   284→//   plumbing; suppressed crossings counted); S reset to 0 on a trip event.
   285→class CusumGuard {
   286→  constructor(adm, cfg) { // cfg: { h, drift, sigmaFloor, sigmaFlat, guardRounds }
   287→    this.adm = adm; this.cfg = cfg;
   288→    this.devs = new Map();      // id -> [{t, d}] — the sender's OWN |p - pooled| stream
   289→    this.base = new Map();      // id -> { mu, sigma, src, n } — honest-floor calibration
   290→    this.S = new Map();         // id -> current CUSUM statistic
   291→    this.track = new Map();     // id -> { n, maxS, maxSat, lastS } — per-sender statistic telemetry
   292→    this.w1Path = [];           // SLEEP's full {t, r, z, S} path (mechanism receipt)
   293→    this.suppressed = new Map();// id -> crossings of h suppressed by an open guard window
   294→    this.guarded = new Map();   // id -> { from, until } — epsNew throwback window
   295→    this.events = [];
   296→    this.guardRounds = new Map();
   297→  }
   298→  rec(id, t, d) {
   299→    if (!this.devs.has(id)) this.devs.set(id, []);
   300→    const a = this.devs.get(id);
   301→    a.push({ t, d });
   302→    if (!this.base.has(id)) {
   303→      const r = this.adm.admittedRound(id);
   304→      if (r === 1) { // founder: no probation exists — its OWN first admitWindow rounds are the reference
   305→        if (a.length === this.adm.admitWindow) {
   306→          const w = a.slice(0, this.adm.admitWindow).map((x) => x.d);
   307→          const s = sd(w);
   308→          const flat = s < this.cfg.sigmaFloor;
   309→          this.base.set(id, { mu: mean(w), sigma: flat ? this.cfg.sigmaFlat : s, src: flat ? 'first40-flat' : 'first40', n: w.length });
   310→        }
   311→      } else if (r != null && this.adm.admitted(id)) { // newcomer: the quantity admitErr judged, verbatim from Admission
   312→        const dm = this.adm.devMean(id);
   313→        if (dm != null) {
   314→          const w = a.filter((x) => x.t < r - 1).map((x) => x.d); // strictly before the admission round = devMean's rounds
   315→          const s = sd(w);
   316→          const flat = s < this.cfg.sigmaFloor;
   317→          this.base.set(id, { mu: dm, sigma: flat ? this.cfg.sigmaFlat : s, src: flat ? 'probation-flat' : 'probation', n: w.length });
   318→        }
   319→      }
   320→    }
   321→  }
   322→  note(t, murmurs, pool) { // post-pool: record devs + CUSUM update + trip check (effective next round)
   323→    for (const m of murmurs) this.rec(m.from, t, Math.abs(+m.p - pool));
   324→    for (const m of murmurs) {
   325→      const id = m.from;
   326→      const g = this.guarded.get(id);
   327→      if (g) {
   328→        if (t > g.until) { this.events.push({ t, id, kind: 'release' }); this.guarded.delete(id); }
   329→        // statistic keeps streaming while guarded (spec formula); only the EVENT is gated
   330→      }
   331→      const b = this.base.get(id);
   332→      if (!b || !this.adm.admitted(id)) continue;
   333→      const r = Math.abs(+m.p - pool);
   334→      const z = (r - b.mu) / b.sigma;
   335→      const Sprev = this.S.get(id) ?? 0;
   336→      const S = Math.max(0, Sprev + z - this.cfg.drift);
   337→      this.S.set(id, S);
   338→      const tr = this.track.get(id) ?? { n: 0, maxS: 0, maxSat: t, lastS: 0 };
   339→      tr.n++; tr.lastS = S;
   340→      if (S > tr.maxS) { tr.maxS = S; tr.maxSat = t; }
   341→      this.track.set(id, tr);
   342→      if (id === SLEEP) this.w1Path.push({ t, r: r6(r), z: r6(z), S: r6(S) });
   343→      if (g) { // guard open: no trip event (E37 plumbing); count suppressed crossings
   344→        if (S > this.cfg.h) this.suppressed.set(id, (this.suppressed.get(id) ?? 0) + 1);
   345→        continue;
   346→      }
   347→      if (S > this.cfg.h) {
   348→        this.S.set(id, 0); // reset on trip (receipted spec)
   349→        const re = this.events.some((e) => e.id === id && (e.kind === 'throw' || e.kind === 'rethrow'));
   350→        this.events.push({
   351→          t, id, kind: re ? 'rethrow' : 'throw', S: r6(S), z: r6(z), r: r6(r),
   352→          mu: r6(b.mu), sigma: r6(b.sigma), baseSrc: b.src,
   353→          trail30: r6(mean(this.devs.get(id).slice(-30).map((x) => x.d))), refMean: r6(b.mu),
   354→        });
   355→        this.guarded.set(id, { from: t + 1, until: t + this.cfg.guardRounds });
   356→      }
   357→    }
   358→  }
   359→  apply(infl, adm, t) { // epsNew for guarded senders (skip ids adm already probationary)
   360→    for (const [id, g] of this.guarded) {
   361→      if (t >= g.from && t <= g.until && infl.has(id) && !adm.probationary(id)) {
   362→        infl.set(id, infl.get(id) * adm.epsNew);
   363→        this.guardRounds.set(id, (this.guardRounds.get(id) ?? 0) + 1);
   364→      }
   365→    }
   366→    return infl;
   367→  }
   368→  summary() {
   369→    const per = {};
   370→    for (const id of [...this.devs.keys()]) {
   371→      const ev = this.events.filter((e) => e.id === id);
   372→      const b = this.base.get(id);
   373→      const tr = this.track.get(id) ?? null;
   374→      per[id] = {
   375→        throws: ev.filter((e) => e.kind === 'throw').length,
   376→        rethrows: ev.filter((e) => e.kind === 'rethrow').length,
   377→        guardRounds: this.guardRounds.get(id) ?? 0,
   378→        mu: b ? r6(b.mu) : null, sigma: b ? r6(b.sigma) : null, baseSrc: b ? b.src : null, baseN: b ? b.n : null,
   379→        nEval: tr ? tr.n : 0, maxS: tr ? r6(tr.maxS) : null, maxSat: tr ? tr.maxSat : null, lastS: tr ? r6(tr.lastS) : null,
   380→        suppressedCrossings: this.suppressed.get(id) ?? 0,
   381→      };
   382→    }
   383→    return { h: this.cfg.h, drift: this.cfg.drift, per, events: this.events.map((e) => ({ ...e })), w1Path: this.w1Path.map((x) => ({ ...x })) };
   384→  }

   387→// TrustReprobation — E37 VERBATIM (unmodified): drains the detector's event
   388→// log; on every throw/rethrow, decay that sender's HedgeTrust weight to
   389→// frac x its current value via absorb(), then let the NORMAL hedge update path
   390→// re-earn it (fixed-share + exp(eta*r)). Recovery is measured per event against
   391→// 0.95 x pre-trip share within `horizon` rounds (R3).
   392→class TrustReprobation {
   393→  constructor(fg, trust, cfg) {
   394→    this.fg = fg; this.trust = trust; this.cfg = cfg;
   395→    this.cursor = 0;   // next detector event index to drain
   396→    this.events = [];  // decay receipts
   397→  }
   398→  process(t) { // after def.note(), before trust.update — decay first, re-earn after
   399→    while (this.cursor < this.fg.events.length) {
   400→      const e = this.fg.events[this.cursor++];
   401→      if (e.kind !== 'throw' && e.kind !== 'rethrow') continue;
   402→      const w = this.trust.weights();
   403→      const pre = w.get(e.id) ?? 0;
   404→      w.set(e.id, pre * this.cfg.frac);
   405→      this.trust.absorb(w);
   406→      this.events.push({
   407→        id: e.id, tripT: e.t, decayT: t, kind: e.kind,
   408→        pre: r6(pre), post: r6(this.trust.weight(e.id)),
   409→        trail30: e.trail30, refMean: e.refMean,
   410→        recoveredAt: null, recoveredIn: null, minShare: null, path: [],
   411→      });
   412→    }
   413→  }
   414→  sample(t) { // post-round (after update + absorb): open-event recovery tracking
   415→    for (const ev of this.events) {
   416→      if (ev.recoveredAt !== null || t <= ev.decayT || t > ev.decayT + this.cfg.horizon) continue;
   417→      const s = this.trust.weight(ev.id);
   418→      if (ev.minShare === null || s < ev.minShare) ev.minShare = r6(s);
   419→      if ((t - ev.decayT) % 10 === 0) ev.path.push({ dt: t - ev.decayT, s: r6(s) });
   420→      if (s >= 0.95 * ev.pre) { ev.recoveredAt = t; ev.recoveredIn = t - ev.decayT; }
   421→    }
   422→  }
   423→  summary() {
   424→    return { frac: this.cfg.frac, horizon: this.cfg.horizon, events: this.events.map((e) => ({ ...e, path: e.path.slice() })) };
   425→  }
   426→}

   463→    : null;

   479→      if (!a || t < JOIN) return 0.5;
   480→      return toxV[t];
   481→    }
   482→    if (id === JOINER) return t >= JOIN ? (h2V[t] ?? 0.5) : 0.5;
   483→    return 0.5;
   484→  };
   485→  const onBus = (id, t) => HONEST.includes(id)
   486→    || (id === JOINER && t >= JOIN)
   487→    || (atkOf(id) != null && t >= JOIN);
   488→

   490→    sumErr: 0, err: [], dmg: {}, share: {}, win: {}, attTrustAtFlip: {}, attTrustFinal: {}, convRound: {},
   491→    rec: { [SLEEP]: {}, [INSTANT]: {}, [JOINER]: {} },
   492→    probTrace: [], w1Curve: [],
   493→    verify: { checks: 0, pass: 0, maxDiff: 0 },
   494→  };

   497→    const murmurs = [];
   498→    for (let i = 0; i < N; i++) murmurs.push({ from: `a${i + 1}`, origin: null, p: votes[t][i] });
   499→    if (t >= JOIN) {

   501→      if (atkOf(INSTANT)) murmurs.push({ from: INSTANT, origin: null, p: pOf(INSTANT, t) });
   502→      murmurs.push({ from: JOINER, origin: null, p: pOf(JOINER, t) });
   503→    }
   504→    prov.inspect(murmurs);
   505→    adm.observe(murmurs);
   506→
   507→    // ---- rewards (supervised pool; absent senders get HedgeTrust default 0.5)
   508→    const rew = new Map();
   509→    for (let i = 0; i < N; i++) rew.set(`a${i + 1}`, 1 - Math.abs(votes[t][i] - s[t]));
   510→    for (const m of murmurs) if (!HONEST.includes(m.from)) rew.set(m.from, 1 - Math.abs(m.p - s[t]));
   511→
   512→    const raw = prov.penalize(trust.weights());
   513→    const infl = adm.reattribute(raw, murmurs, prov);
   514→    for (const d of defs) d.apply(infl, adm, t); // influence-layer consequence (epsNew re-probation)
   515→
   516→    // ---- the sheet does the pooled inference (identical formulas) ----

   518→    for (const id of ALL_IDS) await eng.set(`w.${id}`, infl.get(id) ?? 0);

   520→    adm.notePooled(pool);
   521→    for (const d of defs) d.note(t, murmurs, pool);

   525→      const ps = ALL_IDS.map((id) => pOf(id, t));
   526→      const ref = MurmurBus.pool(ps, ALL_IDS.map((id) => infl.get(id) ?? 0));
   527→      const d = Math.abs(ref - pool);

   529→      if (d < TOL) st.verify.pass++;
   530→      if (d > st.verify.maxDiff) st.verify.maxDiff = d;
   531→      if (d >= TOL) throw new Error(`sheet-verify mismatch arm=${A} seed=${seed} t=${t} diff=${d}`);
   532→    }
   533→
   534→    // ---- accounting ----
   535→    const err = Math.abs(pool - s[t]);
   536→    st.sumErr += 1 - err;
   537→    st.err.push(err);
   538→

   540→    for (const id of [SLEEP, INSTANT, JOINER]) {
   541→      const rec = st.rec[id];
   542→      if (rec.admitExpT === undefined && onBus(id, t) && adm.admitted(id)) {

   544→        rec.admRound = adm.admittedRound(id);
   545→        rec.devMean = adm.devMean(id);
   546→        rec.indep = adm.indepCount(id);

   553→    }

   561→        st.win[a.id].to = t + 1;
   562→        const ps = ALL_IDS.map((id) => pOf(id, t));
   563→        const wFull = ALL_IDS.map((id) => infl.get(id) ?? 0);
   564→        const wCf = wFull.map((w, i) => (ALL_IDS[i] === a.id ? 0 : w));
   565→        const poolCf = MurmurBus.pool(ps, wCf);
   566→        (st.dmg[a.id] = st.dmg[a.id] ?? []).push(err - Math.abs(poolCf - s[t]));
   567→        (st.share[a.id] = st.share[a.id] ?? []).push((infl.get(a.id) ?? 0) / ([...infl.values()].reduce((x, y) => x + y, 0) || 1));
   568→      }
   569→    }
   570→
   571→    // attacker trust trajectory (sleeper): pre-update weight at flip + conviction round
   572→    if (wA) {
   573→      const ws = trust.weights();
   574→      const honMed = median(HONEST.map((id) => ws.get(id) ?? 0));

   577→    }
   578→
   579→    // ---- learn ----
   580→    trust.update(rew);
   581→    trust.absorb(prov.penalize(trust.weights()));

   583→    if (t >= JOIN) st.w1Curve.push(r6(trust.weight(SLEEP) ?? 0)); // R4 post-round trust path
   584→  }
   585→

   588→  const wApost = atkOf(SLEEP);
   589→  if (cusumDef && wApost && wApost.flip != null) {

   591→    const post = trips.filter((e) => e.t >= wApost.flip);
   592→    const pre = trips.filter((e) => e.t < wApost.flip);
   593→    w1Trip = {
   594→      flip: wApost.flip,
   595→      nTrips: trips.length, nPostFlip: post.length, nPreFlip: pre.length,
   596→      firstPostFlipT: post.length ? post[0].t : null,

   598→      tripTs: trips.map((e) => e.t),
   599→    };
   600→  }
   601→
   602→  const errPre = st.err.slice(PRE.from, PRE.to);

   604→  return {

   612→    attTrustAtFlip: st.attTrustAtFlip[SLEEP] ?? null,
