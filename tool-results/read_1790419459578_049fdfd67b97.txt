   268→   268→class FlipGuard {
   269→   269→  constructor(adm, cfg) {
   270→   270→    this.adm = adm; this.cfg = cfg;
   271→   271→    this.devs = new Map();      // id -> [{t, d}] — the sender's OWN |p - pooled| stream
   272→   272→    this.ewma = new Map();      // id -> EWMA_lambda(dev) — the receipted regime tracker
   273→   273→    this.ref = new Map();       // id -> { mean, src } — its OWN probation devMean
   274→   274→    this.guarded = new Map();   // id -> { from, until } — epsNew throwback window
   275→   275→    this.events = [];
   276→   276→    this.guardRounds = new Map();
   277→   277→  }
   278→   278→  rec(id, t, d) {
   279→   279→    if (!this.devs.has(id)) this.devs.set(id, []);
   280→   280→    const a = this.devs.get(id);
   281→   281→    a.push({ t, d });
   282→   282→    const prev = this.ewma.get(id);
   283→   283→    this.ewma.set(id, prev === undefined ? d : (1 - this.cfg.lambda) * prev + this.cfg.lambda * d);
   284→   284→    if (!this.ref.has(id)) {
   285→   285→      const r = this.adm.admittedRound(id);
   286→   286→      if (r === 1) { // founder: no probation exists — its OWN first admitWindow rounds are the reference
   287→   287→        if (a.length === this.adm.admitWindow) this.ref.set(id, { mean: mean(a.slice(0, this.adm.admitWindow).map((x) => x.d)), src: 'first40' });
   288→   288→      } else if (r != null && this.adm.admitted(id)) { // newcomer: the quantity admitErr judged, verbatim from Admission
   289→   289→        const dm = this.adm.devMean(id);
   290→   290→        if (dm != null) this.ref.set(id, { mean: dm, src: 'probation' });
   291→   291→      }
   292→   292→    }
   293→   293→  }
   294→   294→  note(t, murmurs, pool) { // post-pool: record devs + evaluate triggers (effective next round)
   295→   295→    for (const m of murmurs) this.rec(m.from, t, Math.abs(+m.p - pool));
   296→   296→    for (const m of murmurs) {
   297→   297→      const id = m.from;
   298→   298→      const g = this.guarded.get(id);
   299→   299→      if (g) {
   300→   300→        if (t > g.until) { this.events.push({ t, id, kind: 'release' }); this.guarded.delete(id); }
   301→   301→        continue;
   302→   302→      }
   303→   303→      const ref = this.ref.get(id);
   304→   304→      if (!ref || !this.adm.admitted(id)) continue;
   305→   305→      const a = this.devs.get(id);
   306→   306→      if (a.length < this.cfg.trail) continue;
   307→   307→      const trail = mean(a.slice(-this.cfg.trail).map((x) => x.d));
   308→   308→      if (trail > this.cfg.ratio * ref.mean && trail > this.cfg.floor) {
   309→   309→        this.guarded.set(id, { from: t + 1, until: t + this.cfg.guardRounds });
   310→   310→        const re = this.events.some((e) => e.id === id && (e.kind === 'throw' || e.kind === 'rethrow'));
   311→   311→        this.events.push({ t, id, kind: re ? 'rethrow' : 'throw', trail30: r6(trail), refMean: r6(ref.mean), refSrc: ref.src, ewma: r6(this.ewma.get(id)) });
   312→   312→      }
   313→   313→    }
   314→   314→  }
   315→   315→  apply(infl, adm, t) { // epsNew for guarded senders (skip ids adm already probationary)
   316→   316→    for (const [id, g] of this.guarded) {
   317→   317→      if (t >= g.from && t <= g.until && infl.has(id) && !adm.probationary(id)) {
   318→   318→        infl.set(id, infl.get(id) * adm.epsNew);
   319→   319→        this.guardRounds.set(id, (this.guardRounds.get(id) ?? 0) + 1);
   320→   320→      }
   321→   321→    }
   322→   322→    return infl;
   323→   323→  }
   324→   324→  summary() {
   325→   325→    const per = {};
   326→   326→    for (const id of [...this.devs.keys()]) {
   327→   327→      const ev = this.events.filter((e) => e.id === id);
   328→   328→      const ref = this.ref.get(id);
   329→   329→      per[id] = {
   330→   330→        throws: ev.filter((e) => e.kind === 'throw').length,
   331→   331→        rethrows: ev.filter((e) => e.kind === 'rethrow').length,
   332→   332→        guardRounds: this.guardRounds.get(id) ?? 0,
   333→   333→        refMean: ref ? r6(ref.mean) : null, refSrc: ref ? ref.src : null,
   334→   334→      };
   335→   335→    }
   336→   336→    return { per, events: this.events.map((e) => ({ ...e })) };
   337→   337→  }

   340→   340→class Sunset {
   341→   341→  constructor(adm, cfg) {
   342→   342→    this.adm = adm; this.cfg = cfg;
   343→   343→    this.devs = new Map();      // id -> [{t, d}] rolling record (pruned to the window)
   344→   344→    this.member = new Map();    // id -> { nextExpiry } (experiment rounds)
   345→   345→    this.prob = new Map();      // id -> { until } — wrapper probation after a failed re-admission
   346→   346→    this.events = [];
   347→   347→    this.probRounds = new Map();
   348→   348→  }
   349→   349→  note(t, murmurs, pool) { // recording only; re-admission checks run in apply (Admission's own lag)
   350→   350→    for (const m of murmurs) {
   351→   351→      if (!this.devs.has(m.from)) this.devs.set(m.from, []);
   352→   352→      const a = this.devs.get(m.from);
   353→   353→      a.push({ t, d: Math.abs(+m.p - pool) });
   354→   354→      if (a.length > this.cfg.window + 1) a.shift();
   355→   355→    }
   356→   356→  }
   357→   357→  // the admission check re-run over the LAST cfg.window rounds (ending t-1 —
   358→   358→  // the same data-through-t-1 lag Admission's own check uses)
   359→   359→  eval(id, t) {
   360→   360→    const a = (this.devs.get(id) ?? []).filter((x) => x.t >= t - this.cfg.window && x.t < t);
   361→   361→    const m = a.length ? mean(a.map((x) => x.d)) : Infinity;
   362→   362→    const indep = (this.adm.edges.get(id) ?? []).filter((e) => e.indep && e.t > this.adm.round - this.cfg.window).length;
   363→   363→    return { pass: a.length >= this.cfg.window && m <= this.adm.admitErr && indep >= this.adm.minEdgesIndep, mean: m, indep };
   364→   364→  }
   365→   365→  apply(infl, adm, t) {
   366→   366→    for (const id of infl.keys()) {
   367→   367→      if (adm.admittedRound(id) === 1) continue; // founders exempt (D1: constitutive crowd)
   368→   368→      if (!adm.admitted(id)) continue;           // still in adm's own probation — clock not started
   369→   369→      if (this.prob.has(id)) {
   370→   370→        const P = this.prob.get(id);
   371→   371→        if (t >= P.until) {
   372→   372→          const ev = this.eval(id, t);
   373→   373→          if (ev.pass) {
   374→   374→            this.events.push({ t, id, kind: 'readmit-pass', winMean: r6(ev.mean), winIndep: ev.indep });
   375→   375→            this.prob.delete(id);
   376→   376→            this.member.set(id, { nextExpiry: t + this.cfg.every });
   377→   377→          } else {
   378→   378→            this.events.push({ t, id, kind: 'readmit-fail', winMean: r6(ev.mean), winIndep: ev.indep });
   379→   379→            P.until = t + this.cfg.window;
   380→   380→          }
   381→   381→        }
   382→   382→      } else {
   383→   383→        if (!this.member.has(id)) this.member.set(id, { nextExpiry: (adm.admittedRound(id) - 1) + this.cfg.every });
   384→   384→        const M = this.member.get(id);
   385→   385→        if (adm.age(id) > this.cfg.exemptAge) { // incumbent exemption: age > 200
   386→   386→          if (!M.exempt) { M.exempt = true; this.events.push({ t, id, kind: 'exempt-age', age: adm.age(id) }); }
   387→   387→          continue;
   388→   388→        }
   389→   389→        if (t >= M.nextExpiry) {
   390→   390→          const ev = this.eval(id, t);
   391→   391→          if (ev.pass) {
   392→   392→            this.events.push({ t, id, kind: 'sunset-pass', winMean: r6(ev.mean), winIndep: ev.indep });
   393→   393→            M.nextExpiry = t + this.cfg.every;
   394→   394→          } else {
   395→   395→            this.events.push({ t, id, kind: 'sunset-fail', winMean: r6(ev.mean), winIndep: ev.indep });
   396→   396→            this.member.delete(id);
   397→   397→            this.prob.set(id, { until: t + this.cfg.window });
   398→   398→          }
   399→   399→        }
   400→   400→      }
   401→   401→      const P = this.prob.get(id);
   402→   402→      if (P && t <= P.until && !adm.probationary(id)) { // epsNew while lapsed (no double with adm)
   403→   403→        infl.set(id, infl.get(id) * adm.epsNew);
   404→   404→        this.probRounds.set(id, (this.probRounds.get(id) ?? 0) + 1);
   405→   405→      }
   406→   406→    }
   407→   407→    return infl;
   408→   408→  }
   409→   409→  summary() {
   410→   410→    const per = {};
   411→   411→    for (const id of [...this.devs.keys()]) {
   412→   412→      const ev = this.events.filter((e) => e.id === id);
   413→   413→      per[id] = {
   414→   414→        sunsetPass: ev.filter((e) => e.kind === 'sunset-pass').length,
   415→   415→        sunsetFail: ev.filter((e) => e.kind === 'sunset-fail').length,
   416→   416→        readmitPass: ev.filter((e) => e.kind === 'readmit-pass').length,
   417→   417→        readmitFail: ev.filter((e) => e.kind === 'readmit-fail').length,
   418→   418→        probRounds: this.probRounds.get(id) ?? 0,
   419→   419→        exemptAge: ev.some((e) => e.kind === 'exempt-age') || undefined,
   420→   420→      };
   421→   421→    }
   422→   422→    return { per, events: this.events.map((e) => ({ ...e })) };
   423→   423→  }
   424→   424→}

   426→   426→class VelocityBar {
   427→   427→  constructor(adm, cfg) {
   428→   428→    this.adm = adm; this.cfg = cfg;
   429→   429→    this.devs = new Map();    // id -> [{t, d}] full probation dev stream (post-admission kept for receipts)
   430→   430→    this.state = new Map();   // id -> 'admitted' | 'blocked' | undefined (undecided)
   431→   431→    this.events = [];
   432→   432→    this.stats = new Map();
   433→   433→    this.blockRounds = new Map();
   434→   434→  }
   435→   435→  note(t, murmurs, pool) { // recording only; the bar is evaluated AT admission (apply), Admission's own timing
   436→   436→    for (const m of murmurs) {
   437→   437→      if (!this.devs.has(m.from)) this.devs.set(m.from, []);
   438→   438→      this.devs.get(m.from).push({ t, d: Math.abs(+m.p - pool) });
   439→   439→    }
   440→   440→  }
   441→   441→  apply(infl, adm, t) {
   442→   442→    for (const id of infl.keys()) {
   443→   443→      if (adm.admittedRound(id) === 1) { this.state.set(id, 'admitted'); continue; } // founders: no probation, no bar
   444→   444→      let stt = this.state.get(id);
   445→   445→      if (!stt && adm.admitted(id)) {
   446→   446→        // Admission just admitted this sender (its own mean-dev + independence
   447→   447→        // bars passed) — the wrapper adds the velocity clause over the sender's
   448→   448→        // WHOLE probation dev stream: max single-round |p - pooled| <= 1.5*admitErr.
   449→   449→        const a = (this.devs.get(id) ?? []).filter((x) => x.t < t);
   450→   450→        const cumMax = a.length ? Math.max(...a.map((x) => x.d)) : 0;
   451→   451→        const cumMean = a.length ? mean(a.map((x) => x.d)) : Infinity;
   452→   452→        const bar = this.cfg.mult * adm.admitErr;
   453→   453→        stt = cumMax > bar ? 'blocked' : 'admitted';
   454→   454→        this.state.set(id, stt);
   455→   455→        this.stats.set(id, { decidedT: t, probationRounds: a.length, cumMean: r6(cumMean), cumMax: r6(cumMax), bar: r6(bar) });
   456→   456→        this.events.push({ t, id, kind: stt === 'blocked' ? 'vel-block' : 'vel-admit', cumMean: r6(cumMean), cumMax: r6(cumMax), bar: r6(bar) });
   457→   457→      }
   458→   458→      if (stt === 'blocked' && !adm.probationary(id)) { // adm says admitted; the bar says no — epsNew forever
   459→   459→        infl.set(id, infl.get(id) * adm.epsNew);
   460→   460→        this.blockRounds.set(id, (this.blockRounds.get(id) ?? 0) + 1);
   461→   461→      }
   462→   462→    }
   463→   463→    return infl;
   464→   464→  }
   465→   465→  summary() {
   466→   466→    const per = {};
   467→   467→    for (const id of [...this.devs.keys()]) {
   468→   468→      const s = this.stats.get(id);
   469→   469→      per[id] = {
   470→   470→        state: this.state.get(id) ?? 'undecided',
   471→   471→        blockRounds: this.blockRounds.get(id) ?? 0,
   472→   472→        cumMean: s ? s.cumMean : null, cumMax: s ? s.cumMax : null, bar: s ? s.bar : null,
   473→   473→        decidedT: s ? s.decidedT : null, probationRounds: s ? s.probationRounds : null,
   474→   474→      };
   475→   475→    }
   476→   476→    return { per, events: this.events.map((e) => ({ ...e })) };
   477→   477→  }

   505→   505→// ---------------- one seed, paired arms ----------------
   506→   506→async function runArm(seed, world, spec, eng) {
   507→   507→  const { s, votes, toxV, sleepH, h2V } = world;
   508→   508→  const A = spec.name;
   509→   509→  const trust = new HedgeTrust(ALL_IDS, CFG);
   510→   510→  const prov = new Provenance({});
   511→   511→  const adm = new Admission(ADM);
   512→   512→  const def = spec.defense ? new DEF_CLASSES[spec.defense](adm, DEF[spec.defense]) : null;
   513→   513→  const atkOf = (id) => spec.attackers.find((a) => a.id === id) ?? null;
   514→   514→
   515→   515→  // per-round value of a voice (absent -> 0.5; attacker flips at its flip round)
   516→   516→  const pOf = (id, t) => {
   517→   517→    if (HONEST.includes(id)) return votes[t][Number(id.slice(1)) - 1];
   518→   518→    if (id === SLEEP) {
   519→   519→      const a = atkOf(id);
   520→   520→      if (!a || t < JOIN) return 0.5;
   521→   521→      if (a.flip != null && t >= a.flip) return toxV[t];
   522→   522→      return sleepH[t] ?? 0.5;
   523→   523→    }
   524→   524→    if (id === INSTANT) {
   525→   525→      const a = atkOf(id);
   526→   526→      if (!a || t < JOIN) return 0.5;
   527→   527→      return toxV[t];
   528→   528→    }
   529→   529→    if (id === JOINER) return t >= JOIN ? (h2V[t] ?? 0.5) : 0.5;
   530→   530→    return 0.5;
   531→   531→  };
   532→   532→  const onBus = (id, t) => HONEST.includes(id)
   533→   533→    || (id === JOINER && t >= JOIN)
   534→   534→    || (atkOf(id) != null && t >= JOIN);
   535→   535→
   536→   536→  const st = {
   537→   537→    sumErr: 0, err: [], dmg: {}, share: {}, win: {}, attTrustAtFlip: {}, attTrustFinal: {}, convRound: {},
   538→   538→    honW: {}, wAtt: {},
   539→   539→    rec: { [SLEEP]: {}, [INSTANT]: {}, [JOINER]: {} },
   540→   540→    probTrace: [],
   541→   541→    verify: { checks: 0, pass: 0, maxDiff: 0 },
   542→   542→  };
   543→   543→
   544→   544→  for (let t = 0; t < T; t++) {
   545→   545→    const murmurs = [];
   546→   546→    for (let i = 0; i < N; i++) murmurs.push({ from: `a${i + 1}`, origin: null, p: votes[t][i] });
   547→   547→    if (t >= JOIN) {
   548→   548→      if (atkOf(SLEEP)) murmurs.push({ from: SLEEP, origin: null, p: pOf(SLEEP, t) });
   549→   549→      if (atkOf(INSTANT)) murmurs.push({ from: INSTANT, origin: null, p: pOf(INSTANT, t) });
   550→   550→      murmurs.push({ from: JOINER, origin: null, p: pOf(JOINER, t) });
   551→   551→    }
   552→   552→    prov.inspect(murmurs);
   553→   553→    adm.observe(murmurs);
   554→   554→
   555→   555→    // ---- rewards (supervised pool; absent senders get HedgeTrust default 0.5)
   556→   556→    const rew = new Map();
   557→   557→    for (let i = 0; i < N; i++) rew.set(`a${i + 1}`, 1 - Math.abs(votes[t][i] - s[t]));
   558→   558→    for (const m of murmurs) if (!HONEST.includes(m.from)) rew.set(m.from, 1 - Math.abs(m.p - s[t]));
   559→   559→
   560→   560→    const raw = prov.penalize(trust.weights());
   561→   561→    const infl = adm.reattribute(raw, murmurs, prov);
   562→   562→    if (def) def.apply(infl, adm, t); // E36 wrappers: epsNew at the influence layer (no trust contact)
   563→   563→
   564→   564→    // ---- the sheet does the pooled inference (identical formulas) ----
   565→   565→    for (const id of ALL_IDS) await eng.set(`v.${id}`, pOf(id, t));
   566→   566→    for (const id of ALL_IDS) await eng.set(`w.${id}`, infl.get(id) ?? 0);
   567→   567→    const pool = (await eng.get('pool.hedge')).data;
   568→   568→    adm.notePooled(pool);
   569→   569→    if (def) def.note(t, murmurs, pool);
   570→   570→
   571→   571→    if (t % 20 === 0) {
   572→   572→      const ps = ALL_IDS.map((id) => pOf(id, t));
   573→   573→      const ref = MurmurBus.pool(ps, ALL_IDS.map((id) => infl.get(id) ?? 0));
   574→   574→      const d = Math.abs(ref - pool);
   575→   575→      st.verify.checks++;
   576→   576→      if (d < TOL) st.verify.pass++;
   577→   577→      if (d > st.verify.maxDiff) st.verify.maxDiff = d;
   578→   578→      if (d >= TOL) throw new Error(`sheet-verify mismatch arm=${A} seed=${seed} t=${t} diff=${d}`);
   579→   579→    }
   580→   580→
   581→   581→    // ---- accounting ----
   582→   582→    const err = Math.abs(pool - s[t]);
   583→   583→    st.sumErr += 1 - err;
   584→   584→    st.err.push(err);
   585→   585→
   586→   586→    // admission records: FIRST round each newcomer is adm-admitted
   587→   587→    for (const id of [SLEEP, INSTANT, JOINER]) {
   588→   588→      const rec = st.rec[id];
   589→   589→      if (rec.admitExpT === undefined && onBus(id, t) && adm.admitted(id)) {
   590→   590→        rec.admitExpT = t;
   591→   591→        rec.admRound = adm.admittedRound(id);
   592→   592→        rec.devMean = adm.devMean(id);
   593→   593→        rec.indep = adm.indepCount(id);
   594→   594→      }
   595→   595→    }
   596→   596→    // E35-verbatim probation trace for the sleeper (while adm-probationary)
   597→   597→    const wA = atkOf(SLEEP);
   598→   598→    if (wA && t >= JOIN && t < (wA.flip ?? T) && adm.probationary(SLEEP)) {
   599→   599→      st.probTrace.push(Math.abs(pOf(SLEEP, t) - pool));
   600→   600→    }
   601→   601→
   602→   602→    // damage windows: per-attacker counterfactual (zero THAT attacker only)
   603→   603→    for (const a of spec.attackers) {
   604→   604→      if (a.flip == null) continue;
   605→   605→      const from = a.flip, to = Math.min(a.flip + DW, T);
   606→   606→      if (t >= from && t < to) {
   607→   607→        if (st.win[a.id] === undefined) st.win[a.id] = { from: t, to: t + 1 };
   608→   608→        st.win[a.id].to = t + 1;
   609→   609→        const ps = ALL_IDS.map((id) => pOf(id, t));
   610→   610→        const wFull = ALL_IDS.map((id) => infl.get(id) ?? 0);
   611→   611→        const wCf = wFull.map((w, i) => (ALL_IDS[i] === a.id ? 0 : w));
   612→   612→        const poolCf = MurmurBus.pool(ps, wCf);
   613→   613→        (st.dmg[a.id] = st.dmg[a.id] ?? []).push(err - Math.abs(poolCf - s[t]));
   614→   614→        (st.share[a.id] = st.share[a.id] ?? []).push((infl.get(a.id) ?? 0) / ([...infl.values()].reduce((x, y) => x + y, 0) || 1));
   615→   615→      }
   616→   616→    }
   617→   617→
   618→   618→    // attacker trust trajectory (sleeper) + conviction round
   619→   619→    if (wA) {
   620→   620→      const ws = trust.weights();
   621→   621→      const honMed = median(HONEST.map((id) => ws.get(id) ?? 0));
   622→   622→      if (wA.flip != null && t === wA.flip) st.attTrustAtFlip[SLEEP] = r6(ws.get(SLEEP) ?? 0);
   623→   623→      if (st.convRound[SLEEP] === undefined && honMed > 0 && (ws.get(SLEEP) ?? 0) < 0.25 * honMed) st.convRound[SLEEP] = t;
   624→   624→      if ([0, 149, 189, 213, 239, 289, 339, 399].includes(t)) st.wAtt[t] = r6(ws.get(SLEEP) ?? 0);
   625→   625→      if ([0, 149, 189, 213, 239, 289, 339, 399].includes(t)) st.honW[t] = r6(mean(HONEST.map((id) => ws.get(id) ?? 0)));
   626→   626→    }
   627→   627→
   628→   628→    // ---- learn ----
   629→   629→    trust.update(rew);
   630→   630→    trust.absorb(prov.penalize(trust.weights()));
   631→   631→  }
   632→   632→
   633→   633→  const errPre = st.err.slice(PRE.from, PRE.to);
   634→   634→  const dmgOf = (id) => (st.dmg[id] ? mean(st.dmg[id]) : null);
   635→   635→  return {
   636→   636→    name: A,
   637→   637→    poolAcc: st.sumErr / T,
   638→   638→    errPre: mean(errPre),
   639→   639→    dmgW1: dmgOf(SLEEP), dmgG1: dmgOf(INSTANT),
   640→   640→    winW1: st.win[SLEEP] ?? null, winG1: st.win[INSTANT] ?? null,
   641→   641→    shareW1: st.share[SLEEP] ? mean(st.share[SLEEP]) : null,
   642→   642→    shareG1: st.share[INSTANT] ? mean(st.share[INSTANT]) : null,
   643→   643→    attTrustAtFlip: st.attTrustAtFlip[SLEEP] ?? null,
   644→   644→    attTrustFinal: st.attTrustFinal[SLEEP] ?? r6(trust.weight(SLEEP)),
