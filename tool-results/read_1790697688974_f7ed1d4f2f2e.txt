  1088→function collectDepthAmp(seedU32, depth, wLen, atOffsets, preTicks, amp) {
  1089→  const world = new World2(seedU32);
  1090→  const jepa = new Jepa3(seedU32);
  1091→  jepa.lr = LR3; jepa.tau = TAU4;
  1092→  const jInit = new Jepa3(seedU32);
  1093→  let obs = world.observe();
  1094→  for (let t = 0; t < depth; t++) { world.step(); const o1 = world.observe(); jepa.trainStep(obs, o1); obs = o1; }
  1095→  const wStart = depth + 1;
  1096→  const AT = atOffsets.map((o) => wStart + o);
  1097→  const S = []; const Spre = []; const T = []; const T0 = [];
  1098→  const tFrom = wStart - preTicks;
  1099→  let obsPrev = obs;
  1100→  for (let t = tFrom; t <= wStart + wLen - 1; t++) {
  1101→    world.step();
  1102→    const obsNew = world.observe();
  1103→    if (AT.includes(t)) {
  1104→      const px = (world.bx + 8) % (GRID - 2), py = (world.by + 8) % (GRID - 2);
  1105→      for (let dy = 0; dy < 2; dy++) for (let dx = 0; dx < 2; dx++) {
  1106→        obsNew[(py + dy) * GRID + px + dx] = Math.min(1, obsNew[(py + dy) * GRID + px + dx] + amp);
  1107→      }
  1108→    }
  1109→    const zCtx = jepa.encode(obsPrev);
  1110→    const zPred = jepa.predict(zCtx);
  1111→    const zTgt = jepa.encodeTarget(obsNew);
  1112→    const zTgt0 = jInit.encodeTarget(obsNew);
  1113→    const pc = Array.from(perCellSurprise(zPred, zTgt));
  1114→    if (t < wStart) Spre.push(pc); else S.push(pc);
  1115→    if (t >= wStart) { T.push(Array.from(zTgt)); T0.push(Array.from(zTgt0)); }
  1116→    obsPrev = obsNew;
  1117→  }
  1118→  return { S, Spre, T, T0, AT, wStart };
  1119→}
