    88→// ── INDEPENDENT reference stack (no shared code with the sheet) ────────────
    89→const refSma = (c, w) => {
    90→  const out = Array(c.length).fill(null); const pre = [0];
    91→  for (let i = 0; i < c.length; i++) pre.push(pre[i] + c[i]);
    92→  for (let i = w - 1; i < c.length; i++) out[i] = (pre[i + 1] - pre[i + 1 - w]) / w;
    93→  return out;
    94→};
    95→const refRsi = (c, n) => {
    96→  const out = Array(c.length).fill(null); const d = [];
    97→  for (let i = 1; i < c.length; i++) d.push(c[i] - c[i - 1]);
    98→  let ag = 0, al = 0;
    99→  for (let i = 0; i < n; i++) { ag += Math.max(d[i], 0); al += Math.max(-d[i], 0); }
   100→  ag /= n; al /= n; out[n] = al === 0 ? 100 : 100 - 100 / (1 + ag / al);
   101→  for (let i = n; i < d.length; i++) {
   102→    ag = (ag * (n - 1) + Math.max(d[i], 0)) / n;
   103→    al = (al * (n - 1) + Math.max(-d[i], 0)) / n;
   104→    out[i + 1] = al === 0 ? 100 : 100 - 100 / (1 + ag / al);
   105→  }
   106→  return out;
   107→};
   108→const refAtr = (h, l, c, n) => {
   109→  const out = Array(c.length).fill(null);
   110→  const tr = (i) => Math.max(h[i] - l[i], Math.abs(h[i] - c[i - 1]), Math.abs(l[i] - c[i - 1]));
   111→  const ts = [];
   112→  for (let i = 1; i < c.length; i++) ts.push(tr(i));
   113→  let a = 0;
   114→  for (let i = 0; i < n; i++) a += ts[i];
   115→  a /= n; out[n] = a;
   116→  for (let i = n; i < ts.length; i++) { a = (a * (n - 1) + ts[i]) / n; out[i + 1] = a; }
   117→  return out;
   118→};
   119→const refMomz = (c, n, zw) => {
   120→  const mm = Array(c.length).fill(null);
   121→  for (let i = n; i < c.length; i++) mm[i] = c[i] / c[i - n] - 1;
   122→  const out = Array(c.length).fill(null);
   123→  for (let i = n + zw - 1; i < c.length; i++) {
   124→    const win = mm.slice(i - zw + 1, i + 1);
   125→    const m = win.reduce((x, y) => x + y, 0) / zw;
   126→    const sd = Math.sqrt(win.reduce((x, y) => x + (y - m) ** 2, 0) / zw);
   127→    out[i] = sd > 1e-12 ? (mm[i] - m) / sd : 0;
   128→  }
   129→  return out;
   130→};
   131→const refVolz = (h, l, c, n, zw) => {
   132→  const a = refAtr(h, l, c, n);
   133→  const dd = Array(c.length).fill(null);
   134→  for (let i = 1; i < c.length; i++) if (a[i] != null && a[i - 1] != null && a[i - 1] > 0) dd[i] = a[i] / a[i - 1] - 1;
   135→  const out = Array(c.length).fill(null);
   136→  for (let i = n + zw; i < c.length; i++) {
   137→    const win = []; let good = true;
   138→    for (let k = 0; k < zw; k++) { if (dd[i - k] == null) { good = false; break; } win.push(dd[i - k]); }
   139→    if (!good) continue;
   140→    const m = win.reduce((x, y) => x + y, 0) / zw;
   141→    const sd = Math.sqrt(win.reduce((x, y) => x + (y - m) ** 2, 0) / zw);
   142→    out[i] = sd > 1e-12 ? (dd[i] - m) / sd : 0;
   143→  }
   144→  return out;
   145→};
   146→const refWave = (c) => {
   147→  const N = c.length;
   148→  const x = [];
   149→  for (let t = 1; t < N; t++) x.push(c[t] / c[t - 1] - 1);
   150→  const mean = x.reduce((a, b) => a + b, 0) / x.length;
   151→  const bins = [];
   152→  for (let p = 8; p <= 64; p++) {
   153→    let re = 0, im = 0;
   154→    for (let t = 0; t < x.length; t++) {
   155→      const ang = ((2 * Math.PI * (t + 1)) / p) % (2 * Math.PI);
   156→      const xs = x[t] - mean;
   157→      re += xs * Math.cos(ang); im -= xs * Math.sin(ang);
   158→    }
   159→    bins.push({ p, power: (re * re + im * im) / x.length });
   160→  }
   161→  bins.sort((a, b) => b.power - a.power);
   162→  const pStar = bins[0].p;
   163→  let mp = 0;
   164→  for (const b of bins) mp += b.power;
   165→  mp /= bins.length;
   166→  return { p_star: pStar, ratio: bins[0].power / mp };

   168→const refPolicy = (i, c, f, s, r, m, w, wave) => {
   169→  const ind = (j) => f[j] != null && s[j] != null && r[j] != null && m[j] != null;
   170→  const voteT = (j) => {
   171→    if (f[j] == null || s[j] == null || m[j] == null) return 0;
   172→    const gp = f[j] / s[j] - 1;
   173→    return (gp > w.trend_gap && m[j] > w.mom_min) ? 1 : (gp < -w.trend_gap && m[j] < -w.mom_min) ? -1 : 0;
   174→  };
   175→  const voteC = (j) => {
   176→    if (r[j] == null) return 0;
   177→    const tau = 0.35 + w.spec_gate * 0.55;
   178→    const pw = -Math.cos(2 * Math.PI * (j + 1) / wave.p_star + wave.phase);
   179→    return (r[j] <= w.rsi_lo && pw < -tau) ? 1 : (r[j] >= w.rsi_hi && pw > tau) ? -1 : 0;
   180→  };
   181→  if (!ind(i)) return { side: 'FLAT', block: 'none', strength: 0, trendVote: 0, cycleVote: 0 };
   182→  const t0 = voteT(i), t1 = voteT(i - 1), c0 = voteC(i), c1 = voteC(i - 1);
   183→  const tNew = t0 !== 0 && t0 !== t1;
   184→  const cNew = c0 !== 0 && c0 !== c1;
   185→  if (tNew && cNew && t0 !== c0) return { side: t0 > 0 ? 'LONG' : 'SHORT', block: 'conflict', strength: 0.5, trendVote: 1, cycleVote: 1 };
   186→  if (tNew) return { side: t0 > 0 ? 'LONG' : 'SHORT', block: cNew ? 'trend+cycle' : 'trend', strength: cNew ? 1 : 0.5, trendVote: 1, cycleVote: cNew ? 1 : 0 };
   187→  if (cNew) return { side: c0 > 0 ? 'LONG' : 'SHORT', block: 'cycle', strength: 0.5, trendVote: 0, cycleVote: 1 };
   188→  return { side: 'FLAT', block: 'none', strength: 0, trendVote: 0, cycleVote: 0 };
   189→};
   190→const refSim = (ohlcv, w, wave, entConflict) => {
   191→  const c = ohlcv.c, h = ohlcv.h, l = ohlcv.l, N = c.length;
   192→  const f = refSma(c, 10), s = refSma(c, 30), r = refRsi(c, 14), a = refAtr(h, l, c, 14), m = refMomz(c, 20, 30), vz = refVolz(h, l, c, 14, 20);
   193→  const FEE = 0.0006;
   194→  // settlement view: cash + units*price, with units re-derived each bar so the
   195→  // invested fraction of CURRENT equity stays constant — provably the same
   196→  // recursion as the sheet's eq *= (1 + pos*ret), written from the cash side.
   197→  let cash = 1, units = 0, frac = 0, entryEq = 1;
   198→  let entry = 0, entryAtr = 0, openBar = -1;
   199→  const equity = [1], trades = [], changes = [], rets = [];
   200→  let peak = 1, maxdd = 0, held = 0, refusals = 0, ddHalts = 0;
   201→  for (let i = 31; i < N; i++) {
   202→    let eq = cash + units * c[i];       // mark with the units held from last bar
   203→    rets.push(frac !== 0 ? frac * (c[i] / c[i - 1] - 1) : 0);
   204→    if (frac !== 0) held += 1;
   205→    peak = Math.max(peak, eq);
   206→    const ddNow = 1 - eq / peak;
   207→    if (ddNow > maxdd) maxdd = ddNow;
   208→    if (frac !== 0) {
   209→      const dir = Math.sign(frac);
   210→      const stopPx = entry - dir * w.stop_atr * entryAtr;
   211→      const tpPx = entry + dir * w.tp_atr * entryAtr;
   212→      let reason = null;
   213→      if (dir > 0 ? c[i] <= stopPx : c[i] >= stopPx) reason = 'stop';
   214→      else if (dir > 0 ? c[i] >= tpPx : c[i] <= tpPx) reason = 'take-profit';
   215→      else if (i - openBar > 3 * wave.p_star) reason = 'cycle-expiry';
   216→      if (reason) {
   217→        const eqAfterFee = eq * (1 - FEE * Math.abs(frac));
   218→        trades.push({ i_in: openBar, i_out: i, side: dir > 0 ? 'LONG' : 'SHORT', pnl: eqAfterFee / entryEq - 1, reason });
   219→        eq = eqAfterFee; frac = 0; units = 0; changes.push(i);
   220→      }
   221→    }
   222→    if (frac === 0) {
   223→      const prop = refPolicy(i, c, f, s, r, m, w, wave);
   224→      if (prop.side !== 'FLAT') {
   225→        const veto =
   226→          (ddNow > 0.12 ? 'dd' : null) ||
   227→          (changes.slice(-20).length >= 8 ? 'ot' : null) ||
   228→          (vz[i] != null && vz[i] > w.vol_cap ? 'vol' : null) ||
   229→          (prop.block === 'conflict' && entConflict ? 'ent' : null) ||
   230→          (prop.block.indexOf('trend') >= 0 && prop.side === 'SHORT' && wave.regime === 'TREND_UP' && prop.strength < 0.8 ? 'wave' : null);
   231→        if (veto) { refusals++; if (veto === 'dd') ddHalts++; }
   232→        else {
   233→          const dir = prop.side === 'LONG' ? 1 : -1;
   234→          const eqAfterFee = eq * (1 - FEE * w.size);
   235→          entryEq = eqAfterFee;
   236→          frac = dir * w.size;
   237→          units = (frac * eqAfterFee) / c[i];   // invest frac of CURRENT equity
   238→          entry = c[i]; entryAtr = a[i]; openBar = i;
   239→          eq = eqAfterFee; changes.push(i);
   240→        }
   241→      }
   242→    } else {
   243→      units = (frac * eq) / c[i];         // rebalance: constant fraction of equity
   244→    }
   245→    cash = eq - units * c[i];
   246→    equity.push(eq);
   247→  }
   248→  const mean = rets.reduce((x, y) => x + y, 0) / rets.length;
   249→  const va = Math.sqrt(rets.reduce((x, y) => x + (y - mean) ** 2, 0) / rets.length);
   250→  const wins = trades.filter((t) => t.pnl > 0).length;
   251→  return {
   252→    equity, trades,
   253→    metrics: {
   254→      total_return: equity[equity.length - 1] - 1,
   255→      sharpe: va > 1e-12 ? (mean / va) * Math.sqrt(252) : 0,
   256→      maxdd, win_rate: trades.length ? wins / trades.length : 0,
   257→      n_trades: trades.length, exposure: held / (N - 31),
   258→      gate_refusals: refusals, dd_halts: ddHalts,
   259→      score: 0,
   260→    },
   261→  };
   262→};

   266→const clamp = (x, lo, hi) => Math.max(lo, Math.min(hi, x));
   267→const corr = (a, b) => {
   268→  const n = Math.min(a.length, b.length);
   269→  let ma = 0, mb = 0;
   270→  for (let i = 0; i < n; i++) { ma += a[i]; mb += b[i]; }
   271→  ma /= n; mb /= n;
   272→  let sab = 0, sa = 0, sb = 0;
   273→  for (let i = 0; i < n; i++) { sab += (a[i] - ma) * (b[i] - mb); sa += (a[i] - ma) ** 2; sb += (b[i] - mb) ** 2; }
   274→  return sab / Math.sqrt((sa * sb) || 1e-12);

   277→async function harvestEntropy(engine, key, journal, waveCell, nHarvests) {
   278→  const ohlcv = (await engine.get('mkt.ohlcv')).data;
   279→  const c = ohlcv.c;
   280→  const pool = [];
   281→  let sent = waveCell.sent.length ? waveCell.sent : null;
   282→  let got = null, drift = waveCell.drift || 0, jobId = waveCell.job_id, mock = waveCell.mock;
   283→  for (let j = 0; j < nHarvests; j++) {
   284→    // 32 amplitudes of the return waveform, scaled into [0.05, 0.95]
   285→    const vals = [];
   286→    for (let i = 0; i < 32; i++) {
   287→      const t0 = Math.floor((i * (c.length - 1)) / 32);
   288→      const t1 = Math.floor(((i + 1) * (c.length - 1)) / 32);
   289→      let ret = 0;
   290→      for (let t = t0 + 1; t <= Math.max(t1, t0 + 1) && t < c.length; t++) ret += c[t] / c[t - 1] - 1;
   291→      vals.push(clamp(0.5 + ret * 8, 0.05, 0.95));
   292→    }
   293→    if (key) {
   294→      const r = await waveformRead(vals, 2048, key);
   295→      if (r.ok) {
   296→        sent = r.sent; got = r.got;
   297→        drift = got.slice(0, vals.length).reduce((a, v, i) => a + Math.abs(v - vals[i]), 0) / vals.length;
   298→        jobId = r.jobId; mock = false;
   299→        journal.push({ engine: 'qpixl-v1', job_id: r.jobId, what: 'waveform+entropy', ms: r.ms, mock: false });
   300→        for (let i = 0; i < vals.length; i++) pool.push(clamp(0.5 + (got[i] - vals[i]) * 25, 0, 0.999999));
   301→        continue;
   302→      }
   303→      journal.push({ engine: 'qpixl-v1', job_id: null, what: 'waveform+entropy FAILED: ' + r.error, ms: 0, mock: true });
   304→    }
   305→    // offline / fallback: synthetic entropy, FLAGGED
   306→    const rnd = mulberry32(9901 + j * 77);
   307→    sent = vals; got = vals.map((v) => clamp(v + (rnd() - 0.5) * 0.045, 0, 1));
   308→    drift = 0.021; jobId = null; mock = true;
   309→    for (let i = 0; i < vals.length; i++) pool.push(clamp(0.5 + (got[i] - vals[i]) * 25, 0, 0.999999));
   310→  }
   311→  return { pool, wave: { sent, got, drift, job_id: jobId, mock }, journal };
   312→}

   314→async function entangleSnapshot(engine, key, journal, tag) {
   315→  // program cells must be CALLED from the driver (get only sees evaluated cache)
   316→  const g = async (id) => (await engine.call(id)).data;
   317→  const spec = await g('wave.spectrum');
   318→  const rg = await g('wave.regime');
   319→  const momz = await g('ind.momz');
   320→  const rsi = await g('ind.rsi');
   321→  const volz = await g('ind.volz');
   322→  const c = (await engine.get('mkt.ohlcv')).data.c;
   323→  const last = c.length - 1;
   324→  const feat = {
   325→    mom: clamp(Math.abs(momz[last] ?? 0) / 4, 0, 1),
   326→    trend: clamp(spec.r2, 0, 1),
   327→    rev: clamp(Math.abs((rsi[last] ?? 50) - 50) / 50, 0, 1),
   328→    vol: clamp(Math.abs(volz[last] ?? 0) / 4, 0, 1),
   329→  };
   330→  const z = (x) => 2 * x - 1;
   331→  const operations = [
   332→    { type: 'bloch', qubit: 0, paulis: { Z: +z(feat.mom).toFixed(4) } },
   333→    { type: 'bloch', qubit: 1, paulis: { Z: +z(feat.trend).toFixed(4) } },
   334→    { type: 'bloch', qubit: 2, paulis: { Z: +z(feat.rev).toFixed(4) } },
   335→    { type: 'bloch', qubit: 3, paulis: { Z: +z(feat.vol).toFixed(4) } },
   336→    { type: 'relationship', qubits: [0, 1], paulis: { ZZ: 0.6 } },   // mom x trend: reinforce
   337→    { type: 'relationship', qubits: [0, 2], paulis: { ZZ: -0.7 } },  // mom x rev: conflict hypothesis
   338→    { type: 'relationship', qubits: [1, 3], paulis: { ZZ: -0.4 } },
   339→    { type: 'relationship', qubits: [2, 3], paulis: { ZZ: 0.3 } },
   340→  ];
   341→  const coupling = [[0, 1], [0, 2], [1, 3], [2, 3]];
   342→  if (key) {
   343→    const r = await entangleRead(4, coupling, operations, 512, key);
   344→    if (r.ok) {
   345→      journal.push({ engine: 'graph-v1', job_id: r.jobId, what: 'entangle read ' + tag, ms: r.ms, mock: false });
   346→      const rel = (r.tomography && r.tomography.relationships) || {};
   347→      const zzOf = (pair) => {
   348→        const o = rel[pair];
   349→        if (o && typeof o === 'object') return typeof o.ZZ === 'number' ? o.ZZ : 0;
   350→        return num(o);
   351→      };
   352→      const edges = [
   353→        { a: 'mom', b: 'trend', hypo: 0.6, zz: zzOf('0,1') },
   354→        { a: 'mom', b: 'rev', hypo: -0.7, zz: zzOf('0,2') },
   355→        { a: 'trend', b: 'vol', hypo: -0.4, zz: zzOf('1,3') },
   356→        { a: 'rev', b: 'vol', hypo: 0.3, zz: zzOf('2,3') },
   357→      ];
   358→      return { features: feat, edges, agreement: r.agreement, dominant: r.dominant, job_id: r.jobId, mock: false, measurements: r.measurements.slice(0, 5) };
   359→    }
   360→    journal.push({ engine: 'graph-v1', job_id: null, what: 'entangle read ' + tag + ' FAILED: ' + r.error, ms: 0, mock: true });
   361→  }
   362→  // offline fallback: classical pseudo-tomography, FLAGGED
   363→  const rnd = mulberry32(4242 + tag.length);
