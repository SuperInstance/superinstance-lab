   162→const cardArxivs = (blob) => new Set([

   166→function minedMatch(c, hist) {
   167→  const blob = `${c.title} ${c.ref}`.toLowerCase();
   168→  const arxivs = cardArxivs(blob);
   169→  for (const mk of hist.markers) {
   170→    if (mk.slugs.some((s) => blob.includes(s))) return mk.id;
   171→    if (mk.arxivs.some((a) => arxivs.has(a))) return mk.id;
   172→  }
   173→  for (const pd of hist.previously_drawn) if (normTitle(pd.title) === normTitle(c.title)) return `drawn@${pd.run}`;
   174→  return null;
   175→}

   177→// ---------- extract pipeline port — VERBATIM from slice-1 ----------
   178→function extract(corpus, KEYS, hist) {
   179→  const cands = [];
   180→  for (const { src, it } of corpus.items) {
   181→    const blob = JSON.stringify(it).toLowerCase();
   182→    const score = KEYS.reduce((a, k) => a + (blob.includes(k) ? 1 : 0), 0);
   183→    if (score >= 2) {
   184→      cands.push({
   185→        id: `c${cands.length + 1}`,
   186→        source: src,
   187→        title: (it.title || it.full_name || '').slice(0, 140),
   188→        ref: it.url || it.id || `https://github.com/${it.full_name}`,
   189→        stars: it.stars ?? it.points ?? null,
   190→        date: it.created || it.published || it.created_at || null,
   191→        key_score: score,
   192→        gist: (it.desc || it.summary || '').slice(0, 240),
   193→      });
   194→    }
   195→  }
   196→  const sorted = [...cands].sort((a, b) => b.key_score - a.key_score); // stable
   197→  const seen = new Map();
   198→  for (const c of sorted) {
   199→    const k = c.title.toLowerCase().replace(/[^a-z0-9]/g, '').slice(0, 60);
   200→    if (k && !seen.has(k)) seen.set(k, c);
   201→  }
   202→  const pre = [...seen.values()].slice(0, 9);
   203→  const excluded = [];
   204→  const survivors = [];
   205→  for (const c of pre) {
   206→    const m = minedMatch(c, hist);
   207→    if (m) { excluded.push({ id: c.id, title: c.title, source: c.source, ref: c.ref, matched: m }); continue; }
   208→    survivors.push({ ...c, id: `c${survivors.length + 1}` });
   209→  }
   210→  return { pool: cands.length, pre: pre.length, excluded, survivors };
   211→}

   213→// ---------- metric — VERBATIM from slice-1 ----------
   214→function metric(corpora, KEYS, hist) {
   215→  let sum = 0, totalF = 0;
   216→  const per = [];
   217→  for (const c of corpora) {
   218→    const r = extract(c, KEYS, hist);
   219→    const mu = r.survivors.length ? r.survivors.reduce((a, s) => a + s.key_score, 0) / r.survivors.length : 0;
   220→    const f = r.survivors.length;
   221→    sum += mu + f; totalF += f;
   222→    per.push({ corpus: c.name, mu, f, pool: r.pool, pre: r.pre, excluded: r.excluded.length, survivors: r.survivors.map((s) => ({ id: s.id, key_score: s.key_score, title: s.title, ref: s.ref })) });
   223→  }
   224→  return { M: sum / corpora.length, totalF, per_corpus: per };
   225→}

   242→    // slice-1's gate assumed the LIVE mines ledger still equals the run-6-time
   243→    // ledger (11 rows, M1-M11). Mid-session, fleet-seeds engine run-7 (commit
   244→    // 2bc52a5) appended M12 — a legitimate append-only ledger event that
   245→    // slice-1's markers_n==11 equality could never survive. The amended check
   246→    // proves MORE than the original: (a) the committed run-6 marker-id
   247→    // sequence is a PREFIX of the live sequence (append-only, nothing
   248→    // rewritten), (b) exactly the appended ids are recorded, and the
   249→    // substantive replay-fidelity checks are unchanged: exclusions equal +
   250→    // survivors BYTE-EQUAL committed run-6 under the live state.
   251→    const committedIds = committedHist.mined_markers.map((m) => m.id);
   252→    const derivedIds = hist.markers.map((m) => m.id);
   253→    const ledgerPrefixOk = derivedIds.length >= committedIds.length && committedIds.every((id, i) => derivedIds[i] === id);
   254→    const appendedIds = derivedIds.slice(committedIds.length);
   255→    const histEqual =
   256→      ledgerPrefixOk &&
   257→      JSON.stringify(derivedHist.previously_drawn) === JSON.stringify(committedHist.previously_drawn.map((p) => `${p.run}:${p.id}`));
   258→    const exclEqual = JSON.stringify(r.excluded.map((e) => [e.title, e.ref, e.matched])) ===
   259→      JSON.stringify(committedHist.excluded.map((e) => [e.title, e.ref, e.matched]));
   260→    const bytesEqual = canonCandidates(r.survivors) === committed;
   261→    out.run6 = {
   262→      history_state_matches_committed: histEqual,
   263→      history_ledger_append_only_prefix: ledgerPrefixOk,
   264→      history_ledger_committed_rows: committedIds.length,
   265→      history_ledger_live_rows: derivedIds.length,
   266→      history_ledger_appended_marker_ids: appendedIds,
   267→      exclusions_match_committed: exclEqual,
   268→      survivors_byte_equal_committed: bytesEqual,
   269→      derived: derivedHist,
