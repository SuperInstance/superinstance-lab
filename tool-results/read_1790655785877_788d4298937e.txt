   152→function refResponse(handle, tokens, judgedOp, kind, fieldTable) {
   153→  const ph = judgeHash(handle.hash, tokens).probs;
   154→  const pq = judgeQthe(handle.qthe, tokens).probs;
   155→  let probs;
   156→  if (kind === 'fallback_aware') {
   157→    const refField = new SenseTablePrior(JSON.parse(canonicalJSON(fieldTable)));
   158→    probs = refField.observed.has(tokens.join('|'))
   159→      ? refField.priorCounted(tokens.join('|'))
   160→      : ensemble(ph, pq, 0.5); // the v1 law on unseen contexts
   161→  } else {
   162→    probs = ensemble(ph, pq, 0.5);
   163→  }
   164→  const top = argmax(probs);
   165→  const claimed = judgedOp ?? top.op;
   166→  const answers = {
   167→    q1: { type: 'choice', choice: top.op, confidence: top.p, probabilities: probs },
   168→    q2: { type: 'noul', noul: probs[claimed] ?? 0, confidence: top.p, probabilities: probs },
   169→  };
   170→  return {
   171→    model: `jev-garden/${handle.artifact.schema}@w${handle.artifact.index}`,
   172→    answers, escalate: top.p < ESCALATE_BELOW,
   173→    usage: { model: 'garden-local', tokens: 0 },
   174→  };
   175→}
