   105→   105→function horizonRun5(WorldCls, seedU32, lr, tau, wd, T, stopRule, ckpts, captureAt) {

   121→   121→  const postDWindow = stopRule === 'registered' ? 2000 : 0;
   122→   122→  const weightState = () => {
   123→   123→    let anyFin = false;
   124→   124→    for (const W of [jepa.Wc, jepa.Wp, jepa.Wt]) {
   125→   125→      for (let i = 0; i < W.length; i++) { if (Number.isFinite(W[i])) { anyFin = true; break; } }
   126→   126→      if (anyFin) break;
   127→   127→    }
   128→   128→    return { anyFin };
   129→   129→  };
   130→   130→  for (let t = 1; t <= limit; t++) {
   131→   131→    world.step();

   134→   134→    obs = o1;
   135→   135→    losses.push(L);
   136→   136→    executedTo = t;
   137→   137→    if (Number.isFinite(L)) {
   138→   138→      if (D > 0) lossFiniteAgain++;
   139→   139→      else {
   140→   140→        ringWt.push(Float32Array.from(jepa.Wt)); ringWc.push(Float32Array.from(jepa.Wc)); ringWp.push(Float32Array.from(jepa.Wp));
   141→   141→        if (ringWt.length > 201) { ringWt.shift(); ringWc.shift(); ringWp.shift(); }
   142→   142→        if (ckpts.includes(t)) {
   143→   143→          checkpoints[String(t)] = {
   144→   144→            wt_move_rel: relDisp(jepa.Wt, wt0), wc_move_rel: relDisp(jepa.Wc, wc0),
   145→   145→            wt_norm: norm64(jepa.Wt), wc_norm: norm64(jepa.Wc), wp_norm: norm64(jepa.Wp),
   146→   146→            loss_mean_500: mean(losses.slice(Math.max(0, t - 500), t))
   147→   147→          };
   148→   148→        }
   149→   149→        if (captureSet.has(t)) {
   150→   150→          snapshots[String(t)] = { Wc: Float32Array.from(jepa.Wc), Wp: Float32Array.from(jepa.Wp), Wt: Float32Array.from(jepa.Wt) };
   151→   151→        }
   152→   152→      }

   158→   158→      const ws = weightState();

   160→   160→      else if (allNanStep > 0) weightsAbsorbing = false;
   161→   161→      if (t >= D + postDWindow) break;
   162→   162→    }
   163→   163→  }
   164→   164→  const finOrNull = (x) => (Number.isFinite(x) ? x : null);
   165→   165→  const finite = losses.filter(Number.isFinite);
   166→   166→  const out = {
   167→   167→    lr, tau, wd, world: WorldCls === World ? 'trivial' : 'world2',
   168→   168→    T_registered: T, stop_rule: stopRule, cap: stopRule === 'registered' ? null : CONTRAST_CAP,
   169→   169→    executed_to: executedTo,
   170→   170→    D: D > 0 ? D : null, diverged: D > 0,
   171→   171→    D_mod80: D > 0 ? D % 80 : null,
   172→   172→    all_weights_nan_step: D > 0 ? (allNanStep > 0 ? allNanStep : null) : null,
   173→   173→    weights_all_nan_at_end: D > 0 ? !weightState().anyFin : null,
   174→   174→    weights_nan_absorbing: D > 0 ? (allNanStep > 0 && weightsAbsorbing) : null,
   175→   175→    loss_finite_again_count: D > 0 ? lossFiniteAgain : null,
   176→   176→    first10: mean(losses.slice(0, 10)),
   177→   177→    last_finite10_tail: finite.length >= 10 ? mean(finite.slice(-10)) : null,
   178→   178→    loss_ratio_finite_tail: finite.length >= 10 ? mean(finite.slice(-10)) / mean(losses.slice(0, 10)) : null,
   179→   179→    loss_ratio_at_20000: finite.length >= 20000 ? mean(finite.slice(19990, 20000)) / mean(losses.slice(0, 10)) : null,
   180→   180→    checkpoints,
   181→   181→    captured_at: Object.keys(snapshots).map(Number),
   182→   182→    snapshots,
   183→   183→    drift_last200: null, wc_norm_at_last_finite: null, wp_norm_at_last_finite: null,
   184→   184→    blowup_rate_last200: null
   185→   185→  };
   186→   186→  if (D > 201 && ringWt.length === 201) {
   187→   187→    const a = ringWt[0], b = ringWt[200];
   188→   188→    let drift = 0, n2 = 0;
   189→   189→    for (let i = 0; i < a.length; i++) { const d = b[i] - a[i]; drift += d * d; n2 += b[i] * b[i]; }
   190→   190→    out.drift_last200 = finOrNull(Math.sqrt(drift) / Math.sqrt(n2));
   191→   191→    const wa = ringWc[0], wb = ringWc[200];
   192→   192→    out.blowup_rate_last200 = finOrNull(Math.log(norm64(wb) / norm64(wa)) / 200);
   193→   193→    out.wc_norm_at_last_finite = finOrNull(norm64(ringWc[200]));
   194→   194→    out.wp_norm_at_last_finite = finOrNull(norm64(ringWp[200]));
   195→   195→  }
   196→   196→  return out;
