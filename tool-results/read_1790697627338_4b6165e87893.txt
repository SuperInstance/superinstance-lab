   117→function cellFeatures(o, c) {

   124→function solve4(A, B) { // Gauss-Jordan 4x4: A M = B
   125→  const n = 4;
   126→  const Aug = A.map((row, i) => [...row, ...B[i]]);
   127→  for (let col = 0; col < n; col++) {
   128→    let piv = col;
   129→    for (let r = col + 1; r < n; r++) if (Math.abs(Aug[r][col]) > Math.abs(Aug[piv][col])) piv = r;
   130→    [Aug[col], Aug[piv]] = [Aug[piv], Aug[col]];
   131→    const p = Aug[col][col];
   132→    for (let j = 0; j < 2 * n; j++) Aug[col][j] /= p;
   133→    for (let r = 0; r < n; r++) {
   134→      if (r === col) continue;
   135→      const f = Aug[r][col];
   136→      if (f !== 0) for (let j = 0; j < 2 * n; j++) Aug[r][j] -= f * Aug[col][j];
   137→    }
   138→  }
   139→  return Aug.map((row) => row.slice(n));
   140→}

   993→function horizonRun5(WorldCls, seedU32, lr, tau, wd, T, stopRule, ckpts, captureAt) {

  1002→  const captureSet = new Set(captureAt);
  1003→  const snapshots = {};
  1004→  const ringWt = []; const ringWc = []; const ringWp = [];
  1005→  let D = -1, executedTo = 0;
  1006→  let allNanStep = -1, weightsAbsorbing = true, lossFiniteAgain = 0;
  1007→  let obs = world.observe();
  1008→  const limit = stopRule === 'registered' ? T : Math.min(T, CONTRAST_CAP);
  1009→  const postDWindow = stopRule === 'registered' ? 2000 : 0;
  1010→  const weightState = () => {
  1011→    let anyFin = false;
  1012→    for (const W of [jepa.Wc, jepa.Wp, jepa.Wt]) {
  1013→      for (let i = 0; i < W.length; i++) { if (Number.isFinite(W[i])) { anyFin = true; break; } }
  1014→      if (anyFin) break;
  1015→    }

  1017→  };
  1018→  for (let t = 1; t <= limit; t++) {
  1019→    world.step();

  1022→    obs = o1;
  1023→    losses.push(L);
  1024→    executedTo = t;
  1025→    if (Number.isFinite(L)) {
  1026→      if (D > 0) lossFiniteAgain++;
  1027→      else {

  1029→        if (ringWt.length > 201) { ringWt.shift(); ringWc.shift(); ringWp.shift(); }
  1030→        if (ckpts.includes(t)) {

  1032→            wt_move_rel: relDisp(jepa.Wt, wt0), wc_move_rel: relDisp(jepa.Wc, wc0),
  1033→            wt_norm: norm64(jepa.Wt), wc_norm: norm64(jepa.Wc), wp_norm: norm64(jepa.Wp),
  1034→            loss_mean_500: mean(losses.slice(Math.max(0, t - 500), t))
  1035→          };
  1036→        }

  1038→          snapshots[String(t)] = { Wc: Float32Array.from(jepa.Wc), Wp: Float32Array.from(jepa.Wp), Wt: Float32Array.from(jepa.Wt) };
  1039→        }
  1040→      }
  1041→    } else if (D < 0) {
  1042→      D = t;
  1043→      if (stopRule !== 'registered') break;
  1044→    }
  1045→    if (D > 0 && t > D && t <= D + postDWindow) {
  1046→      const ws = weightState();
  1047→      if (!ws.anyFin) { if (allNanStep < 0) allNanStep = t; }
  1048→      else if (allNanStep > 0) weightsAbsorbing = false;
  1049→      if (t >= D + postDWindow) break;
  1050→    }

  1053→  const finite = losses.filter(Number.isFinite);
  1054→  const out = {
  1055→    lr, tau, wd, world: WorldCls === World ? 'trivial' : 'world2',
  1056→    T_registered: T, stop_rule: stopRule, cap: stopRule === 'registered' ? null : CONTRAST_CAP,
  1057→    executed_to: executedTo,
  1058→    D: D > 0 ? D : null, diverged: D > 0,
  1059→    D_mod80: D > 0 ? D % 80 : null,
  1060→    all_weights_nan_step: D > 0 ? (allNanStep > 0 ? allNanStep : null) : null,
  1061→    weights_all_nan_at_end: D > 0 ? !weightState().anyFin : null,
  1062→    weights_nan_absorbing: D > 0 ? (allNanStep > 0 && weightsAbsorbing) : null,

  1064→    first10: mean(losses.slice(0, 10)),

  1067→    loss_ratio_at_20000: finite.length >= 20000 ? mean(finite.slice(19990, 20000)) / mean(losses.slice(0, 10)) : null,
  1068→    checkpoints,
  1069→    captured_at: Object.keys(snapshots).map(Number),
  1070→    snapshots,
  1071→    drift_last200: null, wc_norm_at_last_finite: null, wp_norm_at_last_finite: null,
  1072→    blowup_rate_last200: null
  1073→  };
  1074→  if (D > 201 && ringWt.length === 201) {
  1075→    const a = ringWt[0], b = ringWt[200];
  1076→    let drift = 0, n2 = 0;
  1077→    for (let i = 0; i < a.length; i++) { const d = b[i] - a[i]; drift += d * d; n2 += b[i] * b[i]; }
  1078→    out.drift_last200 = finOrNull(Math.sqrt(drift) / Math.sqrt(n2));
  1079→    const wa = ringWc[0], wb = ringWc[200];
  1080→    out.blowup_rate_last200 = finOrNull(Math.log(norm64(wb) / norm64(wa)) / 200);
  1081→    out.wc_norm_at_last_finite = finOrNull(norm64(ringWc[200]));
  1082→    out.wp_norm_at_last_finite = finOrNull(norm64(ringWp[200]));
  1083→  }
  1084→  return out;
  1085→}

  1104→      const px = (world.bx + 8) % (GRID - 2), py = (world.by + 8) % (GRID - 2);

  1106→        obsNew[(py + dy) * GRID + px + dx] = Math.min(1, obsNew[(py + dy) * GRID + px + dx] + amp);
  1107→      }

  1111→    const zTgt = jepa.encodeTarget(obsNew);
