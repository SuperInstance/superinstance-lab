    85→def file_seal(path: Path):
    86→    b = path.read_bytes()

   110→def check_registration():
   111→    if not REG_PATH.exists():
   112→        print("FATAL: registration.json missing — claims must be registered "
   113→              "BEFORE running conformance: python3 conformance.py --register")
   114→        sys.exit(2)

   117→        cur = file_seal(HERE / f)

   119→            print("FATAL: seal stale for %s (sha256 moved since registration) "
   120→                  "— re-register and re-run" % f)

   123→            print("FATAL: seal stale for %s (mtime moved since registration) "
   124→                  "— re-register and re-run" % f)
   125→            sys.exit(2)

   127→        print("FATAL: vendored micromoth.py sha256 %s != pinned %s — the "
   128→              "oracle moved; re-seal deliberately" % (MICROMOTH_SHA256,
   129→                                                      EXPECTED_MICROMOTH_SHA256))
   130→        sys.exit(2)

   248→def build_circuit(spec):
   249→    cc = CellCircuit(spec["n"], spec["m"], name=spec["name"],
   250→                     shots_plan={"shots": spec["shots"],
   251→                                 "seed_policy": "unsealed" if spec.get(
   252→                                     "unsealed") else "sealed"},
   253→                     simulator_sha256=MICROMOTH_SHA256)
   254→    spec["build"](cc)
   255→    cc.seal()
   256→    cc.proof(precision=12)
   257→    for v in spec["views"]:
   258→        cc.view(v)
   259→    if spec["shots"]:
   260→        cc.sample(spec["shots"], seed=spec["seed"], noise_model=spec.get("noise"))
   261→    if spec.get("forget"):
   262→        cc.forget(shots=spec["forget"]["shots"],
   263→                  reason=spec["forget"]["reason"])
   264→    return cc

   267→# --- tamper machinery -----------------------------------------------------------
   268→
   269→def _flip_char(line: bytes, pos: int) -> bytes:
   270→    c = line[pos:pos + 1]
   271→    ch = c.decode()
   272→    if ch in "0123456789abcdef":
   273→        new = "0123456789abcdef"[(int(ch, 16) + 1) % 16]
   274→    elif "a" <= ch <= "z":
   275→        new = chr((ord(ch) - 97 + 1) % 26 + 97)
   276→    elif ch.isdigit():
   277→        new = str((int(ch) + 1) % 10)
   278→    else:
   279→        new = "x" if ch != "x" else "y"
   280→    return line[:pos] + new.encode() + line[pos + 1:]

   283→def tamper_variants(data: bytes):
   284→    """For every row: three single-byte tampers (id flip, kind flip, prev
   285→    flip — the row-0 prev flip IS the genesis tamper). Yields
   286→    (row_index, variant, tampered_bytes)."""
   287→    lines = data.split(b"\n")[:-1]
   288→    out = []
   289→    for i, line in enumerate(lines):
   290→        for field in ("id", "kind", "prev"):
   291→            marker = ('"%s":"' % field).encode()
   292→            pos = line.index(marker) + len(marker)
   293→            tampered = lines[:]
   294→            tampered[i] = _flip_char(line, pos)
   295→            out.append((i, "%s-flip" % field,
   296→                        b"\n".join(tampered) + b"\n"))
   297→    return out

   300→def _reforge_from(rows, seq, mutate):
   301→    """Full re-forgery: mutate row `seq`'s args, then recompute ids and
   302→    prev links for that row and every row after it (defeats chain
   303→    verification; the anchor and the init payload pin remain)."""
   304→    rows = [json.loads(json.dumps(r)) for r in rows]
   305→    mutate(rows[seq]["args"])
   306→    prev = rows[seq - 1]["id"] if seq > 0 else GENESIS
   307→    for r in rows[seq:]:
   308→        r["prev"] = prev
   309→        body = {k: r[k] for k in ("seq", "op", "kind", "args", "prev")}
   310→        r["id"] = fnv1a64(canon(body))
   311→        prev = r["id"]
   312→    return b"\n".join(canon(r) for r in rows) + b"\n", rows
