   146→const median = (a) => { const b = [...a].sort((x, y) => x - y); const m = b.length >> 1; return b.length % 2 ? b[m] : (b[m - 1] + b[m]) / 2; };

   150→function makeRng(harvest, vault, purpose) {
   151→  let d = 0;

   154→// paired one-sided sign test, zeros excluded (E28 receipted small-p method)
   155→const binom = (n, k) => { let r = 1; for (let i = 0; i < k; i++) r = (r * (n - i)) / (i + 1); return r; };
   156→function signTestOneSided(deltas) {
   157→  const W = deltas.filter((d) => d > 1e-12).length;
   158→  const L = deltas.filter((d) => d < -1e-12).length;
   159→  const n = W + L;
   160→  let p = 0;
   161→  for (let k = W; k <= n; k++) p += binom(n, k);
   162→  return { W, L, n, p: p / 2 ** n };
   163→}

   167→// ---------------- world ----------------
   168→// Ingredients only: the flip timing is applied per arm as a selection over
   169→// these shared series (paired-worlds doctrine — no arm-keyed draws).
   170→function genWorld(seed, harvest, vault) {
   171→  const wR = makeRng(harvest, vault, `e35:world:${seed}`);
   172→  const qR = makeRng(harvest, vault, `e35:skill:${seed}`);
   173→  const xR = makeRng(harvest, vault, `e35:tox:${seed}`);
   174→  const shR = makeRng(harvest, vault, `e35:sleep:${seed}`);
   175→  const nhR = makeRng(harvest, vault, `e35:h2:${seed}`);
   176→  const s = new Array(T);
   177→  s[0] = wR() < 0.5 ? 0 : 1;
   178→  for (let t = 1; t < T; t++) s[t] = wR() < FLIP_P ? 1 - s[t - 1] : s[t - 1];
   179→  // mixed acuity honest founders; a1 is the high-acuity reference voice
   180→  const q = Array.from({ length: T }, () => new Array(N));
   181→  q[0][0] = 0.9;
   182→  for (let i = 1; i < N; i++) q[0][i] = Math.round((0.5 + 0.45 * qR()) * 1000) / 1000;
   183→  for (let t = 1; t < T; t++) {
   184→    for (let i = 0; i < N; i++) {
   185→      q[t][i] = qR() < REROLL_P ? Math.round((0.5 + 0.45 * qR()) * 1000) / 1000 : q[t - 1][i];
   186→    }
   187→  }
   188→  const vR = [];
   189→  for (let i = 0; i < N; i++) vR.push(makeRng(harvest, vault, `e35:vote:${seed}:${i}`));
   190→  const votes = Array.from({ length: T }, () => new Array(N));
   191→  for (let t = 0; t < T; t++) {
   192→    for (let i = 0; i < N; i++) {
   193→      const signal = vR[i]() < q[t][i] ? s[t] : 1 - s[t];
   194→      votes[t][i] = clampP(signal === 1 ? 0.5 + (q[t][i] - 0.5) : 0.5 - (q[t][i] - 0.5));
   195→    }
   196→  }
   197→  // the TOXIC series — E24 formula VERBATIM: stable acuity TOX_Q, never
   198→  // rerolled — a consistent liar whose values point the wrong way ~75% of the
   199→  // time (confidence |TOX_Q-0.5| = 0.25 from neutral). Shared by the sleeper
   200→  // post-flip AND the instant toxic newcomer (same toxic values across arms).
   201→  const toxV = new Array(T);
   202→  for (let t = 0; t < T; t++) {
   203→    const signal = xR() < TOX_Q ? s[t] : 1 - s[t];
   204→    toxV[t] = clampP(signal === 1 ? 0.5 + (TOX_Q - 0.5) : 0.5 - (TOX_Q - 0.5));
   205→  }
   206→  // the SLEEPER's honest phase — e24's h1 generator VERBATIM (expert-range
   207→  // acuity drawn once, iid per-round errors, REROLL_P acuity drift): an
   208→  // honest-looking newcomer whose |p - pooled| matches honest voices, with
   209→  // its OWN edges (no copying — the evasion requires passing the independence
   210→  // bar honestly).
   211→  const sleepH = new Array(T).fill(null);
   212→  {
   213→    let qs = Math.round((0.7 + 0.25 * shR()) * 1000) / 1000;
   214→    for (let t = JOIN; t < T; t++) {
   215→      if (shR() < REROLL_P) qs = Math.round((0.7 + 0.25 * shR()) * 1000) / 1000;
   216→      const signal = shR() < qs ? s[t] : 1 - s[t];
   217→      sleepH[t] = clampP(signal === 1 ? 0.5 + (qs - 0.5) : 0.5 - (qs - 0.5));
   218→    }
   219→  }
   220→  // the honest late-joiner h2 — E21/E24 honest-newmember control, honest forever
   221→  const h2V = new Array(T).fill(null);
   222→  {
   223→    let qh = Math.round((0.7 + 0.25 * nhR()) * 1000) / 1000;
   224→    for (let t = JOIN; t < T; t++) {
   225→      if (nhR() < REROLL_P) qh = Math.round((0.7 + 0.25 * nhR()) * 1000) / 1000;
   226→      const signal = nhR() < qh ? s[t] : 1 - s[t];
   227→      h2V[t] = clampP(signal === 1 ? 0.5 + (qh - 0.5) : 0.5 - (qh - 0.5));
   228→    }
   229→  }
   230→  return { s, votes, toxV, sleepH, h2V };
   231→}

   233→// ---------------- sheet (15 voice slots; formulas IDENTICAL for every arm) --
   234→function buildSheet() {
   235→  const cells = [];
   236→  for (const id of ALL_IDS) cells.push({ id: `v.${id}`, kind: 'value', value: 0.5, description: `voice ${id} (posterior P(s=1))` });
   237→  for (const id of ALL_IDS) cells.push({ id: `w.${id}`, kind: 'value', value: 1 / V, description: `influence weight ${id} (protocol-adjusted)` });
   238→  const lg = (x) => `Math.log(clamp(${x},0.02,0.98)/(1-clamp(${x},0.02,0.98)))`;
   239→  const sig = (z) => `(1/(1+Math.exp(-(${z}))))`;
   240→  const hT = [], hD = [];
   241→  for (const id of ALL_IDS) { hT.push(`w.${id}*${lg(`v.${id}`)}`); hD.push(`w.${id}`); }
   242→  cells.push({ id: 'pool.hedge', kind: 'formula', expr: sig(`(${hT.join(' + ')}) / (${hD.join(' + ')})`) });
   243→  // in-sheet meters: sleeper share; instant toxic share; honest joiner share
   244→  cells.push({ id: 'amp.w1', kind: 'formula', expr: `w.${SLEEP} / (${hD.join(' + ')})` });
   245→  cells.push({ id: 'amp.g1', kind: 'formula', expr: `w.${INSTANT} / (${hD.join(' + ')})` });
   246→  cells.push({ id: 'amp.h2', kind: 'formula', expr: `w.${JOINER} / (${hD.join(' + ')})` });
   247→  return { id: `sleeper-${V}`, title: `E35 sleeper cell (${V} voice slots)`, cells };
   248→}

   265→// ---------------- one seed, paired arms (or one strategy probe arm) --------
   266→// spec: { name, protocol, attacker } — attacker.flip = toxic start round.
   267→async function runArm(seed, world, spec, eng) {
   268→  const { s, votes, toxV, sleepH, h2V } = world;
   269→  const A = spec.name;
   270→  const att = spec.attacker;
   271→  const trust = new HedgeTrust(ALL_IDS, CFG);
   272→  const prov = new Provenance({});
   273→  const adm = spec.protocol === 'v3.1' ? new Admission(ADM) : null;
   274→
   275→  // per-round value of a voice (absent -> 0.5; attacker flips at att.flip)
   276→  const pOf = (id, t) => {
   277→    if (HONEST.includes(id)) return votes[t][Number(id.slice(1)) - 1];
   278→    if (id === SLEEP) {
   279→      if (!att || att.id !== SLEEP) return 0.5;
   280→      if (t < JOIN) return 0.5;
   281→      if (att.flip != null && t >= att.flip) return toxV[t];
   282→      return sleepH[t] ?? 0.5;
   283→    }
   284→    if (id === INSTANT) {
   285→      if (!att || att.id !== INSTANT) return 0.5;
   286→      return t >= JOIN ? toxV[t] : 0.5;
   287→    }
   288→    if (id === JOINER) return t >= JOIN ? (h2V[t] ?? 0.5) : 0.5;
   289→    return 0.5;
   290→  };
   291→  const onBus = (id, t) => HONEST.includes(id)
   292→    || (id === JOINER && t >= JOIN)
   293→    || (att && att.id === id && t >= JOIN);
   294→
   295→  const st = {
   296→    sumErr: 0, err: [], dmg: [], winFrom: null, winTo: null, attShare: [], attShareWin: [],
   297→    attEcho: [], attTagEnd: null, attTrustAtFlip: null, attTrustFinal: null, convRound: null,
   298→    honW: {}, wAtt: {},
   299→    w1: { admitExpT: null, admRound: null, devMean: null, indep: null, probTrace: [] },
   300→    h2: { admitExpT: null, admRound: null, devMean: null, indep: null },
   301→    g1: { admitExpT: null, admRound: null, devMean: null, indep: null },
   302→    verify: { checks: 0, pass: 0, maxDiff: 0 },
   303→  };
   304→
   305→  for (let t = 0; t < T; t++) {
   306→    const murmurs = [];
   307→    for (let i = 0; i < N; i++) murmurs.push({ from: `a${i + 1}`, origin: null, p: votes[t][i] });
   308→    if (t >= JOIN) {
   309→      if (att && att.id === SLEEP) murmurs.push({ from: SLEEP, origin: null, p: pOf(SLEEP, t) });
   310→      if (att && att.id === INSTANT) murmurs.push({ from: INSTANT, origin: null, p: pOf(INSTANT, t) });
   311→      murmurs.push({ from: JOINER, origin: null, p: pOf(JOINER, t) });
   312→    }
   313→    prov.inspect(murmurs);
   314→    if (adm) adm.observe(murmurs);
   315→
   316→    // ---- rewards (supervised pool; absent senders get HedgeTrust default 0.5)
   317→    const rew = new Map();
   318→    for (let i = 0; i < N; i++) rew.set(`a${i + 1}`, 1 - Math.abs(votes[t][i] - s[t]));
   319→    for (const m of murmurs) if (!HONEST.includes(m.from)) rew.set(m.from, 1 - Math.abs(m.p - s[t]));
   320→
   321→    const raw = prov.penalize(trust.weights()); // v3 state penalty (all arms)
   322→    const infl = adm ? adm.reattribute(raw, murmurs, prov) : prov.influence(raw, murmurs);
   323→
   324→    // ---- the sheet does the pooled inference (identical formulas) ----
   325→    for (const id of ALL_IDS) await eng.set(`v.${id}`, pOf(id, t));
   326→    for (const id of ALL_IDS) await eng.set(`w.${id}`, infl.get(id) ?? 0);
   327→    const pool = (await eng.get('pool.hedge')).data;
   328→    if (adm) adm.notePooled(pool);
   329→
   330→    if (t % 20 === 0) {
   331→      const ps = ALL_IDS.map((id) => pOf(id, t));
   332→      const ref = MurmurBus.pool(ps, ALL_IDS.map((id) => infl.get(id) ?? 0));
   333→      const d = Math.abs(ref - pool);
   334→      st.verify.checks++;
   335→      if (d < TOL) st.verify.pass++;
   336→      if (d > st.verify.maxDiff) st.verify.maxDiff = d;
   337→      if (d >= TOL) throw new Error(`sheet-verify mismatch arm=${A} seed=${seed} t=${t} diff=${d}`);
   338→    }
   339→
   340→    // ---- accounting ----
   341→    const err = Math.abs(pool - s[t]);
   342→    st.sumErr += 1 - err;
   343→    st.err.push(err);
   344→
   345→    // admission records (v3.1 arms): capture the FIRST round each newcomer
   346→    // is admitted + the probation deviation trace for the attacker
   347→    if (adm) {
   348→      const cap = (id, rec) => {
   349→        if (rec.admitExpT === null && onBus(id, t) && adm.admitted(id)) {
   350→          rec.admitExpT = t;
   351→          rec.admRound = adm.admittedRound(id);
   352→          rec.devMean = adm.devMean(id);
