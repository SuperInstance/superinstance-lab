   225→function firstBalancedObject(t) {
   226→  // scan from the first '{', tracking string state; cut where depth returns to 0
   227→  const a = t.indexOf('{');
   228→  if (a === -1) return null;
   229→  let depth = 0, inStr = false, esc = false;
   230→  for (let i = a; i < t.length; i++) {
   231→    const c = t[i];
   232→    if (inStr) {
   233→      if (esc) esc = false;
   234→      else if (c === '\\') esc = true;
   235→      else if (c === '"') inStr = false;
   236→      continue;
   237→    }
   238→    if (c === '"') inStr = true;
   239→    else if (c === '{') depth++;
   240→    else if (c === '}') { depth--; if (depth === 0) return { text: t.slice(a, i + 1), repaired: false }; }
   241→  }
   242→  if (!inStr && depth > 0) return { text: t.slice(a) + '}'.repeat(depth), repaired: true }; // unclosed braces at EOF
   243→  return null; // unterminated string or no object: refuse
   244→}

   246→function extractJson(raw) {
   247→  const t = raw.trim();
   248→  try { return { obj: JSON.parse(t), method: 'direct' }; } catch { /* fall through */ }
   249→  const fence = t.match(/```(?:json)?\s*([\s\S]*?)```/);
   250→  if (fence) {
   251→    try { return { obj: JSON.parse(fence[1].trim()), method: 'fence-strip' }; } catch { /* fall through */ }
   252→  }
   253→  const a = t.indexOf('{'), b = t.lastIndexOf('}');
   254→  if (a !== -1 && b > a) {
   255→    try { return { obj: JSON.parse(t.slice(a, b + 1)), method: 'brace-extract' }; } catch { /* fall through */ }
   256→  }
   257→  // declared repairs — raw stays sealed verbatim; only the DERIVED parse is repaired, method named:
   258→  const fbo = firstBalancedObject(t);
   259→  if (fbo) {
   260→    try {
   261→      const obj = JSON.parse(fbo.text);
   262→      return { obj, method: fbo.repaired ? 'brace-close' : 'first-object' };
   263→    } catch { /* fall through */ }
   264→  }
   265→  return { obj: null, method: 'failed' };
   266→}

   268→async function callDeepseek(userMsg, temperature) {
   269→  const key = process.env.DEEPSEEK_API_KEY;
   270→  if (!key) throw new Error('DEEPSEEK_API_KEY missing from env (use: set -a; source .env; set +a)');
   271→  const body = JSON.stringify({
   272→    model: REQUESTED_MODEL,
   273→    messages: [
   274→      { role: 'system', content: SYSTEM_PREFIX },
   275→      { role: 'user', content: userMsg },
   276→    ],
   277→    temperature,
   278→    max_tokens: MAX_TOKENS,
   279→    stream: false,
   280→  });
   281→  let lastErr = null;
   282→  for (let attempt = 1; attempt <= MAX_ATTEMPTS; attempt++) {
   283→    const ctrl = new AbortController();
   284→    const timer = setTimeout(() => ctrl.abort(), CALL_TIMEOUT_MS);
   285→    try {
   286→      const res = await fetch(ENDPOINT, {
   287→        method: 'POST',
   288→        headers: { 'Authorization': `Bearer ${key}`, 'Content-Type': 'application/json' },
   289→        body,
   290→        signal: ctrl.signal,
   291→      });
   292→      clearTimeout(timer);
   293→      if (res.status === 429 || res.status >= 500) {
   294→        const retryAfter = Number(res.headers.get('retry-after') || 0);
   295→        const waitMs = retryAfter > 0 ? retryAfter * 1000 : Math.min(60_000, 1500 * 2 ** (attempt - 1)) + Math.floor(Math.random() * 400);
   296→        lastErr = `http_${res.status}`;
   297→        console.log(`  attempt ${attempt}: ${lastErr}, backing off ${waitMs}ms`);
   298→        await sleep(waitMs);
   299→        continue;
   300→      }
   301→      if (!res.ok) {
   302→        const txt = await res.text();
   303→        throw new Error(`http_${res.status}: ${txt.slice(0, 300)}`);
   304→      }
   305→      const j = await res.json();
   306→      return { ok: true, j };
   307→    } catch (e) {
   308→      clearTimeout(timer);
   309→      lastErr = e.name === 'AbortError' ? `timeout_${CALL_TIMEOUT_MS}ms` : String(e.message || e);
   310→      if (String(lastErr).startsWith('http_4')) throw lastErr; // 4xx (non-429) is not retryable
   311→      if (attempt < MAX_ATTEMPTS) {
   312→        const waitMs = Math.min(60_000, 1500 * 2 ** (attempt - 1)) + Math.floor(Math.random() * 400);
   313→        console.log(`  attempt ${attempt}: ${lastErr}, backing off ${waitMs}ms`);
   314→        await sleep(waitMs);
   315→      }
   316→    }
   317→  }
   318→  return { ok: false, error: lastErr };
   319→}

   331→function usd(usage, p = PRICES.off_peak) {
   332→  const hit = usage.prompt_cache_hit_tokens / 1e6 * p.input_cache_hit_per_1m;
   333→  const miss = usage.prompt_cache_miss_tokens / 1e6 * p.input_cache_miss_per_1m;
   334→  const out = usage.completion_tokens / 1e6 * p.output_per_1m;
   335→  const nocache = (usage.prompt_tokens / 1e6) * p.input_cache_miss_per_1m + out;
   336→  return {
   337→    with_cache: hit + miss + out, if_no_cache: nocache,
   338→    input_with_cache: hit + miss,
   339→    input_if_no_cache: (usage.prompt_tokens / 1e6) * p.input_cache_miss_per_1m,
   340→  };
   341→}

   424→async function cmdReparse() {
   425→  // post-hoc declared parse pass: refreshes content_json/parse_method on sealed rows
   426→  // (content_raw is NEVER touched; a parse_repaired_at field marks rows whose method changed)
   427→  const lines = fs.readFileSync(ROWS_FILE, 'utf8').split('\n').filter(Boolean);
   428→  const out = [];
   429→  let changed = 0;
   430→  for (const line of lines) {
   431→    const row = JSON.parse(line);
   432→    if (row.content_raw !== undefined) {
   433→      const { obj, method } = extractJson(row.content_raw);
   434→      if (method !== row.parse_method || JSON.stringify(obj) !== JSON.stringify(row.content_json)) {
   435→        row.parse_method_prior = row.parse_method ?? null;
   436→        row.parse_method = method;
   437→        row.content_json = obj;
   438→        row.parse_repaired_at = now();
   439→        changed++;
   440→      }
   441→      if (row.usage) {
   442→        row.usd = usd(row.usage); // derived wrapper field recomputed with the current declared price basis
   443→      }
   444→    }
   445→    out.push(JSON.stringify(row));
   446→  }
   447→  fs.writeFileSync(ROWS_FILE, out.join('\n') + '\n', 'utf8');
   448→  // rebuild situations.jsonl from the (repaired) situation rows
   449→  const rows = readRows().filter((r) => r.task === 'situation' && r.content_json);
   450→  const sel = fs.existsSync(SEL_FILE) ? JSON.parse(fs.readFileSync(SEL_FILE, 'utf8')).selected_call_n : [];
   451→  fs.writeFileSync(SITS_FILE, rows.map((r) => JSON.stringify({
   452→    call_id: r.call_id, call_n: r.call_n, question_id: r.question_id,
   453→    situation: r.content_json, parse_method: r.parse_method,
   454→    usage: r.usage, sealed_at: r.sealed_at, model_served: r.model_served,
   455→    recommended_for_EQ6: sel.includes(r.call_n) ? true : (fs.existsSync(SEL_FILE) ? false : null),
   456→    selection_reasons: sel.includes(r.call_n) ? JSON.parse(fs.readFileSync(SEL_FILE, 'utf8')).criteria : null,
   457→  })).join('\n') + '\n', 'utf8');
   458→  writeEconomics();
   459→  console.log(`reparse complete: ${changed} rows re-parsed (methods declared in-row); situations.jsonl rebuilt with ${rows.length} rows`);
   460→}
