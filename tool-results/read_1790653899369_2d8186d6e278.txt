   127→function refResponse(handle, tokens, judgedOp, kind, fieldTable) {
   128→  const ph = judgeHash(handle.hash, tokens).probs;
   129→  const pq = judgeQthe(handle.qthe, tokens).probs;
   130→  let probs;
   131→  if (kind === 'everywhere') {
   132→    const refField = new SenseTablePrior(JSON.parse(canonicalJSON(fieldTable))); // re-parsed serialized aggregate
   133→    probs = refField.priorCounted(tokens.join('|'));
   134→  } else if (kind === 'fallback_aware') {
   135→    const refField = new SenseTablePrior(JSON.parse(canonicalJSON(fieldTable)));
   136→    probs = refField.observed.has(tokens.join('|'))
   137→      ? refField.priorCounted(tokens.join('|'))
   138→      : ensemble(ph, pq, 0.5); // the v1 law on unseen contexts
   139→  } else {
   140→    probs = ensemble(ph, pq, 0.5);
   141→  }
   142→  const top = argmax(probs);
   143→  const claimed = judgedOp ?? top.op;
   144→  const answers = {
   145→    q1: { type: 'choice', choice: top.op, confidence: top.p, probabilities: probs },
   146→    q2: { type: 'noul', noul: probs[claimed] ?? 0, confidence: top.p, probabilities: probs },
   147→  };
   148→  return {
   149→    model: `jev-garden/${handle.artifact.schema}@w${handle.artifact.index}`,
   150→    answers, escalate: top.p < ESCALATE_BELOW,
   151→    usage: { model: 'garden-local', tokens: 0 },
   152→  };
   153→}
